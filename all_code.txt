    private fun processTtsWavTo16kWithBeep(sourceWav: File, context: Context): Boolean {
        if (!sourceWav.exists() || sourceWav.length() < 44) return false
        val fileBytes = sourceWav.readBytes()
        if (fileBytes.size < 44) return false

        var dataOffset = 44
        var dataLen = fileBytes.size - 44
        var sourceSampleRate = 22050
        var sourceChannels = 1
        var bitsPerSample = 16

        var offset = 12
        while (offset < fileBytes.size - 8) {
            val chunkId = String(fileBytes, offset, 4)
            val chunkSize = ByteBuffer.wrap(fileBytes, offset + 4, 4).order(ByteOrder.LITTLE_ENDIAN).int
            if (chunkId == "fmt ") {
                sourceChannels = ByteBuffer.wrap(fileBytes, offset + 10, 2).order(ByteOrder.LITTLE_ENDIAN).short.toInt()
                sourceSampleRate = ByteBuffer.wrap(fileBytes, offset + 12, 4).order(ByteOrder.LITTLE_ENDIAN).int
                bitsPerSample = ByteBuffer.wrap(fileBytes, offset + 22, 2).order(ByteOrder.LITTLE_ENDIAN).short.toInt()
            } else if (chunkId == "data") {
                dataOffset = offset + 8
                dataLen = chunkSize
                break
            }
            offset += 8 + chunkSize
        }

        if (sourceSampleRate <= 0) sourceSampleRate = 22050
        if (sourceChannels <= 0) sourceChannels = 1
        if (bitsPerSample <= 0) bitsPerSample = 16

        val bytesPerSample = bitsPerSample / 8
        val frameSize = bytesPerSample * sourceChannels
        val totalFrames = dataLen / frameSize

        val sourcePcm = ShortArray(totalFrames)
        var byteIdx = dataOffset
        for (i in 0 until totalFrames) {
            if (byteIdx + frameSize > fileBytes.size) break
            var monoSample = 0
            for (ch in 0 until sourceChannels) {
                val s = (fileBytes[byteIdx + ch * 2].toInt() and 0xFF or (fileBytes[byteIdx + ch * 2 + 1].toInt() shl 8)).toShort()
                monoSample += s
            }
            sourcePcm[i] = (monoSample / sourceChannels).toShort()
            byteIdx += frameSize
        }

        val resampledCount = ((sourcePcm.size.toLong() * SAMPLE_RATE_HZ) / sourceSampleRate).toInt()
        val resampledPcm = ShortArray(resampledCount)
        val ratio = sourceSampleRate.toDouble() / SAMPLE_RATE_HZ.toDouble()

        var rawSpeechPeak = 1
        for (i in 0 until resampledCount) {
            val srcPos = i * ratio
            val srcIdx = srcPos.toInt()
            val frac = srcPos - srcIdx
            val s0 = if (srcIdx < sourcePcm.size) sourcePcm[srcIdx] else 0
            val s1 = if (srcIdx + 1 < sourcePcm.size) sourcePcm[srcIdx + 1] else s0
            val interpolated = (s0 * (1.0 - frac) + s1 * frac).toInt().coerceIn(-32768, 32767)
            if (abs(interpolated) > rawSpeechPeak) rawSpeechPeak = abs(interpolated)
            resampledPcm[i] = interpolated.toShort()
        }

        val normGain = (28000.0 / rawSpeechPeak.toDouble()).coerceIn(1.0, 8.0)
        for (i in 0 until resampledCount) {
            val boosted = (resampledPcm[i].toInt() * normGain).toInt().coerceIn(-32768, 32767)
            resampledPcm[i] = boosted.toShort()
        }

        val leadInSamples = ShortArray((SAMPLE_RATE_HZ * 0.200).toInt())
        val beepDurationMs = 500
        val beepSampleCount = (SAMPLE_RATE_HZ * beepDurationMs) / 1000
        val beepSamples = ShortArray(beepSampleCount)
        val beepAmp = 26000.0

        for (i in 0 until beepSampleCount) {
            val attack = if (i < 160) (i / 160.0) else 1.0
            val release = if (i > beepSampleCount - 160) ((beepSampleCount - i) / 160.0) else 1.0
            val angle = 2.0 * PI * i * 800.0 / SAMPLE_RATE_HZ
            val s = (sin(angle) * beepAmp * attack * release).toInt().coerceIn(-32768, 32767)
            beepSamples[i] = s.toShort()
        }

        val pauseSamples = ShortArray((SAMPLE_RATE_HZ * 0.120).toInt())
        val tailSamples = ShortArray((SAMPLE_RATE_HZ * 0.150).toInt())

        val totalFinalSamples = leadInSamples.size + resampledPcm.size + pauseSamples.size + beepSamples.size + tailSamples.size
        val finalPcmBytes = ByteArray(totalFinalSamples * 2)
        val buf = ByteBuffer.wrap(finalPcmBytes).order(ByteOrder.LITTLE_ENDIAN)

        for (s in leadInSamples) buf.putShort(0)
        for (s in resampledPcm) buf.putShort(s)
        for (s in pauseSamples) buf.putShort(0)
        for (s in beepSamples) buf.putShort(s)
        for (s in tailSamples) buf.putShort(0)

        val targetFile = File(context.filesDir, DEFAULT_GREETING_FILE_NAME)
        val dir = File(context.filesDir, GREETING_DIR_NAME).apply { mkdirs() }
        val dirGreeting = File(dir, DEFAULT_GREETING_FILE_NAME)
        val versionMarker = File(dir, GREETING_VERSION_MARKER)

        FileOutputStream(targetFile).use { fos ->
            writeWavHeader(fos, finalPcmBytes.size.toLong(), SAMPLE_RATE_HZ, CHANNELS, BITS_PER_SAMPLE)
            fos.write(finalPcmBytes)
            fos.flush()
        }

        runCatching { targetFile.copyTo(dirGreeting, overwrite = true) }
        runCatching { versionMarker.writeText("v2_normalized") }
        stageToTmpIfPossible(targetFile)
        return targetFile.length() > 64000
    }

    fun generateImmediateFallbackGreeting(outputFile: File) {
        val pcmData = ArrayList<Short>()
        appendSilence(pcmData, 200)
        appendChimeNote(pcmData, 523.25, 180, 0.80)
        appendChimeNote(pcmData, 659.25, 180, 0.80)
        appendChimeNote(pcmData, 783.99, 220, 0.85)
        appendChimeNote(pcmData, 1046.50, 380, 0.85)
        appendSilence(pcmData, 250)

        appendSilence(pcmData, 150)

        val pcmBytes = ByteArray(pcmData.size * 2)
        val byteBuffer = ByteBuffer.wrap(pcmBytes).order(ByteOrder.LITTLE_ENDIAN)
        for (sample in pcmData) byteBuffer.putShort(sample)

        outputFile.parentFile?.mkdirs()
        FileOutputStream(outputFile).use { fos ->
            writeWavHeader(fos, pcmBytes.size.toLong(), SAMPLE_RATE_HZ, CHANNELS, BITS_PER_SAMPLE)
            fos.write(pcmBytes)
            fos.flush()
        }
    }

    private fun appendSilence(buffer: ArrayList<Short>, durationMs: Int) {
        val count = (SAMPLE_RATE_HZ * (durationMs / 1000.0)).toInt()
        for (i in 0 until count) buffer.add(0)
    }

    private fun appendSineTone(buffer: ArrayList<Short>, frequencyHz: Double, durationMs: Int, amplitude: Double) {
        val totalSamples = (SAMPLE_RATE_HZ * (durationMs / 1000.0)).toInt()
        val maxAmp = 32767.0 * amplitude
        for (i in 0 until totalSamples) {
            val angle = 2.0 * PI * i * frequencyHz / SAMPLE_RATE_HZ
            buffer.add((sin(angle) * maxAmp).toInt().coerceIn(-32768, 32767).toShort())
        }
    }

    private fun appendChimeNote(buffer: ArrayList<Short>, frequencyHz: Double, durationMs: Int, peakAmplitude: Double) {
        val totalSamples = (SAMPLE_RATE_HZ * (durationMs / 1000.0)).toInt()
        val maxAmp = 32767.0 * peakAmplitude
        for (i in 0 until totalSamples) {
            val t = i.toDouble() / SAMPLE_RATE_HZ
            val decay = kotlin.math.exp(-3.5 * (i.toDouble() / totalSamples))
            buffer.add((sin(2.0 * PI * frequencyHz * t) * maxAmp * decay).toInt().coerceIn(-32768, 32767).toShort())
        }
    }

    fun writeWavHeader(out: OutputStream, totalAudioLen: Long, sampleRate: Int, channels: Int, bitsPerSample: Int) {
        val totalDataLen = totalAudioLen + 36
        val byteRate = (sampleRate * channels * bitsPerSample / 8).toLong()
        val blockAlign = (channels * bitsPerSample / 8).toShort()
        val header = ByteArray(44)
        val buffer = ByteBuffer.wrap(header).order(ByteOrder.LITTLE_ENDIAN)

        buffer.put('R'.code.toByte()); buffer.put('I'.code.toByte()); buffer.put('F'.code.toByte()); buffer.put('F'.code.toByte())
        buffer.putInt(totalDataLen.toInt())
        buffer.put('W'.code.toByte()); buffer.put('A'.code.toByte()); buffer.put('V'.code.toByte()); buffer.put('E'.code.toByte())
        buffer.put('f'.code.toByte()); buffer.put('m'.code.toByte()); buffer.put('t'.code.toByte()); buffer.put(' '.code.toByte())
        buffer.putInt(16); buffer.putShort(1); buffer.putShort(channels.toShort()); buffer.putInt(sampleRate)
        buffer.putInt(byteRate.toInt()); buffer.putShort(blockAlign); buffer.putShort(bitsPerSample.toShort())
        buffer.put('d'.code.toByte()); buffer.put('a'.code.toByte()); buffer.put('t'.code.toByte()); buffer.put('a'.code.toByte())
        buffer.putInt(totalAudioLen.toInt())
        out.write(header, 0, 44)
    }

                for (t in activePlaybackTracks) {
                    runCatching { t.write(pcmPayload, 0, minOf(trackBufferSize, pcmPayload.size)) }
                }
            }

            var offset = minOf(trackBufferSize, pcmPayload.size)
            while (isSessionActive.get() && isGreetingPlaying.get() && offset < pcmPayload.size) {
                val len = minOf(CHUNK_BYTES, pcmPayload.size - offset)
                if (audioManager.isSpeakerphoneOn) {
                    audioManager.isSpeakerphoneOn = false
                }
                synchronized(activePlaybackTracks) {
                    val iterator = activePlaybackTracks.iterator()
                    while (iterator.hasNext()) {
                        val t = iterator.next()
                        if (t.routedDevice?.type == AudioDeviceInfo.TYPE_BUILTIN_SPEAKER) {
                            runCatching { t.pause(); t.flush(); t.stop(); t.release() }
                            iterator.remove()
                            continue
                        }
                        runCatching { t.write(pcmPayload, offset, len, AudioTrack.WRITE_NON_BLOCKING) }
                    }
                }
                offset += len
                try {
                    Thread.sleep(85)
                } catch (e: InterruptedException) {
                    break
                }
            }

            if (isSessionActive.get() && isGreetingPlaying.get()) {
                try { Thread.sleep(400) } catch (ignored: InterruptedException) {}
            }

            cleanupPlaybackTracks()

            runCatching {
                audioManager.setParameters(QUALCOMM_PARAM_UPLINK_OFF)
                audioManager.setParameters(MEDIATEK_PARAM_UPLINK_OFF)
                audioManager.setParameters(SAMSUNG_PARAM_UPLINK_OFF)
                if (savedMusicVolume >= 0) {
                    audioManager.setStreamVolume(AudioManager.STREAM_MUSIC, savedMusicVolume, 0)
                }
            }

            isGreetingPlaying.set(false)
            if (isSessionActive.get()) {
                onFinished()
            }
        }
    }

                                    .setChannelMask(AudioFormat.CHANNEL_IN_MONO)
                                    .build()
                            )
                            .setBufferSizeInBytes(recordBufferSize)

                            HiddenApiBypass.invoke(
                                AudioRecord.Builder::class.java,
                                recBuilder,
                                "setContext",
                                captureContext
                            )
                        }

                        if (presetApplied) {
                            runCatching {
                                HiddenApiBypass.invoke(
                                    AudioRecord.Builder::class.java,
                                    recBuilder,
                                    "setAudioAttributes",
                                    attrBuilder.build()
                                )
                            }
                        } else if (candidate.sourceId != AUDIO_SOURCE_VOICE_DOWNLINK && candidate.sourceId != AUDIO_SOURCE_VOICE_CALL) {
                            recBuilder.setAudioSource(candidate.sourceId)
                        }

                        val candidateRec = recBuilder.build()
                        if (candidateRec.state == AudioRecord.STATE_INITIALIZED) {
                            record = candidateRec
                        } else {
                            candidateRec.release()
                        }
                    } catch (ignored: Throwable) {}

                    if (record == null) {
                        try {

                            val fallbackRec = AudioRecord(

                                SAMPLE_RATE_HZ,
                                AudioFormat.CHANNEL_IN_MONO,
                                AudioFormat.ENCODING_PCM_16BIT,
                                recordBufferSize
                            )
                            if (fallbackRec.state == AudioRecord.STATE_INITIALIZED) {
                                record = fallbackRec
                            } else {
                                fallbackRec.release()
                            }
                        } catch (ignored: Throwable) {}

                    if (candidate.routeToTelephony && telephonyInput != null && hasPrivilege) {
                        runCatching { finalRec.setPreferredDevice(telephonyInput) }
                    }
                    finalRec.startRecording()
                    if (finalRec.recordingState == AudioRecord.RECORDSTATE_RECORDING) finalRec else {
                        finalRec.release()
                        null
                    }
                } catch (t: Throwable) {
                    null
                }
            }

            var currentCandidateIndex = 0
            var currentRecord: AudioRecord? = null

            while (currentCandidateIndex < candidates.size) {
                currentRecord = createAudioRecord(candidates[currentCandidateIndex])
                if (currentRecord != null) break
                currentCandidateIndex++
            }

                isCaptureActive.set(false)
                return@Thread
            }

            activeAudioRecord = currentRecord
            val rawBuffer = ShortArray(CHUNK_SAMPLES)
            var currentGain = 4.0f

            while (isSessionActive.get() && isCaptureActive.get()) {
                val record = activeAudioRecord ?: break
                if (!hasPrivilege && audioManager.isMicrophoneMute) {
                    audioManager.isMicrophoneMute = false
                }

                val read = record.read(rawBuffer, 0, rawBuffer.size)
                if (read > 0) {
                    var peak = 0
                    for (i in 0 until read) {
                        val absVal = abs(rawBuffer[i].toInt())
                        if (absVal > peak) peak = absVal
                    }

                    if (peak in 50..12000) {
                        val targetGain = (20000.0f / peak).coerceIn(1.0f, 15.0f)
                        currentGain = (currentGain * 0.7f) + (targetGain * 0.3f)
                    }

                    val boosted = ShortArray(read)
                    for (i in 0 until read) {
                        boosted[i] = (rawBuffer[i].toFloat() * currentGain).toInt().coerceIn(-32768, 32767).toShort()
                    }
                    onPcmChunk(boosted)
                } else if (read < 0) {
                    try { Thread.sleep(50) } catch (ignored: Throwable) {}
                }
            }

            runCatching {
                currentRecord?.stop()
                currentRecord?.release()
            }
            activeAudioRecord = null
            isCaptureActive.set(false)
        }, "StandaloneAudioCaptureThread").apply { start() }
    }

        cleanupPlaybackTracks()
        runCatching {
            activeAudioRecord?.stop()
            activeAudioRecord?.release()
        }
        activeAudioRecord = null
        captureThread?.interrupt()
        captureThread = null
    }

        synchronized(activePlaybackTracks) {
            for (track in activePlaybackTracks) {
                runCatching { track.stop(); track.release() }
            }
            activePlaybackTracks.clear()
        }
    }

            .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)

            )
            true
        }.getOrDefault(false)

        if (!applied) {
            builder.setUsage(AudioAttributes.USAGE_VOICE_COMMUNICATION_SIGNALLING)
        }
        return builder.build()
    }

    private fun readWavPayload(file: File): ByteArray {
        if (!file.exists() || file.length() <= 44) {
            return ByteArray(SAMPLE_RATE_HZ * 2 * 3)
        }
        return try {
            FileInputStream(file).use { fis ->
                val fullBytes = fis.readBytes()
                if (fullBytes.size > 44) fullBytes.copyOfRange(44, fullBytes.size) else ByteArray(SAMPLE_RATE_HZ * 2 * 3)
            }
        } catch (t: Throwable) {
            ByteArray(SAMPLE_RATE_HZ * 2 * 3)
        }
    }

            }
        }

     * Reloads all voicemails from database and JSON storage into memory.
     */
    fun reloadFromDisk() {
        repositoryScope.launch {
            try {
                val dbList = dao.getAllVoicemailsSync()
                if (dbList.isNotEmpty()) {
                    _voicemails.value = dbList
                } else {
                    // Check if JSON file exists
                    val jsonFile = File(context.filesDir, JSON_FILE_NAME)
                    if (jsonFile.exists() && jsonFile.length() > 2) {
                        val parsed = parseVoicemailsFromJson(jsonFile)
                        if (parsed.isNotEmpty()) {

                            _voicemails.value = parsed
                        }
                    }

                Log.e(TAG, "Error in reloadFromDisk", t)
            }

        dao.getVoicemailById(id)

        val entityWithId = voicemail.copy(id = insertedId)
        val currentList = _voicemails.value.toMutableList()
        currentList.removeAll { it.id == insertedId }

        currentList.removeAll { it.id == voicemail.id }

            exportToJsonFiles()

        }

                val obj = JSONObject().apply {
                    put("id", c.id)
                    put("callerName", c.callerName ?: "")
                    put("phoneNumber", c.phoneNumber)
                    put("simSlotIndex", c.simSlotIndex)
                    put("simDisplayName", c.simDisplayName)
                    put("timestamp", c.timestamp)
                    put("durationSeconds", c.durationSeconds)
                    put("transcript", c.transcript)
                    put("audioFilePath", c.audioFilePath)
                    put("isRead", c.isRead)
                    put("isPickedUp", c.isPickedUp)
                }
                jsonArray.put(obj)
            }

            val jsonContent = jsonArray.toString(2).toByteArray(Charsets.UTF_8)

            // Write to files/voicemails.json
            val mainFile = File(context.filesDir, JSON_FILE_NAME)
            FileOutputStream(mainFile).use { fos ->
                fos.write(jsonContent)
                fos.flush()
            }

            // Write backup
            val backupFile = File(context.filesDir, JSON_BACKUP_NAME)
            FileOutputStream(backupFile).use { fos ->
                fos.write(jsonContent)
                fos.flush()
            }

            Log.d(TAG, "Exported ${list.size} voicemails to $JSON_FILE_NAME and $JSON_BACKUP_NAME")

                simSlotIndex = obj.optInt("simSlotIndex", 0),
                simDisplayName = obj.optString("simDisplayName", "SIM 1"),
                timestamp = obj.optLong("timestamp", System.currentTimeMillis()),
                durationSeconds = obj.optInt("durationSeconds", 0),
                transcript = obj.optString("transcript", ""),
                audioFilePath = obj.optString("audioFilePath", ""),
                isRead = obj.optBoolean("isRead", false),
                isPickedUp = obj.optBoolean("isPickedUp", false)
            )
            result.add(entity)
        }
        return result
    }

                                    }
                                }
                                Log.d(TAG, "Extracted $assetPath -> ${dest.absolutePath} (${dest.length()} bytes)")
                                copied = true
                                break
                            } catch (e: Exception) {
                                // Try next candidate path
                            }
                        }
                        if (!copied) {
                            Log.w(TAG, "Failed finding/extracting $name from assets in candidates $candidatePaths")
                        }
                    }
                    if (dest.exists() && dest.length() > 0L) {
                        modelFiles[name] = dest
                    }
                }

                val encoderFile = modelFiles["encoder.int8.onnx"]
                val decoderFile = modelFiles["decoder.onnx"]
                val joinerFile = modelFiles["joiner.onnx"]
                val tokensFile = modelFiles["tokens.txt"]

                        decoder = decoderFile.absolutePath,

        pcmChannel = channel

    /**
     * Consumer loop: receives 16kHz PCM chunks, writes raw bytes, feeds Sherpa-ONNX stream.
     * Dynamically instantiates [activeStream] on the fly inside the loop if the model was still
     * initializing when the call was answered.
     */
    private suspend fun consumeAudioChunks(channel: Channel<ShortArray>) {
        var chunkCounter = 0

        for (chunk in channel) {
            if (chunk.isEmpty()) continue

            // 1. Write raw PCM to temporary disk file
            val byteBuffer = ByteBuffer.allocate(chunk.size * 2).order(ByteOrder.LITTLE_ENDIAN)
            for (sample in chunk) {
                val absSample = abs(sample.toInt())
                if (absSample > peakSessionAmplitude) {
                    peakSessionAmplitude = absSample
                }
                byteBuffer.putShort(sample)
            }
            try {
                tempPcmOutputStream?.write(byteBuffer.array(), 0, byteBuffer.position())
            } catch (e: Exception) {
                Log.e(TAG, "Error writing raw PCM chunk to disk", e)
            }

            // 2. Calculate RMS Audio Level for real-time visual waveform animation
            val rms = calculateRms(chunk)
            _audioRmsLevel.value = rms

            chunkCounter++
            if (chunkCounter % 20 == 0) {
                Log.d(TAG, "Transcriber chunk #$chunkCounter processed: samples=${chunk.size}, rms=$rms, sessionPeak=$peakSessionAmplitude")
            }

            // 3. Normalize samples to FloatArray [-1.0f, 1.0f]
            val floatSamples = FloatArray(chunk.size)
            for (i in chunk.indices) {
                floatSamples[i] = chunk[i] / 32768.0f
            }

            // 4. Dynamic model/stream binding: create stream on the fly if model just finished loading!
            val rec = recognizer
            if (rec != null && activeStream == null && !isStopping.get()) {
                activeStream = rec.createStream()
                Log.i(TAG, "Active OnlineStream instantiated dynamically on chunk arrival!")
            }
            val stream = activeStream

            // 5. Feed Sherpa-ONNX OnlineStream
            if (rec != null && stream != null) {
                stream.acceptWaveform(floatSamples, SAMPLE_RATE_HZ)

                while (rec.isReady(stream)) {
                    rec.decode(stream)
                }

                val currentResult = rec.getResult(stream)
                val partialText = currentResult.text.trim()

                if (rec.isEndpoint(stream)) {
                    if (partialText.isNotEmpty()) {
                        if (accumulatedSentences.isNotEmpty()) {
                            accumulatedSentences.append(" ")
                        }
                        accumulatedSentences.append(partialText)
                    }
                    rec.reset(stream)
                    _liveTranscript.value = accumulatedSentences.toString()
                } else {
                    val fullDisplay = if (accumulatedSentences.isEmpty()) {
                        partialText
                    } else if (partialText.isEmpty()) {
                        accumulatedSentences.toString()
                    } else {
                        "$accumulatedSentences $partialText"
                    }
                    _liveTranscript.value = fullDisplay
                }
            } else {
                if (rms > 0.08f) {
                    val current = _liveTranscript.value
                    if (!current.contains("â€¦")) {
                        _liveTranscript.value = if (current.isEmpty()) "Speakingâ€¦" else "$current â€¦"
                    }
                }
            }
        }

        Log.i(TAG, "Consumer coroutine completed chunk processing loop (totalChunks=$chunkCounter, peakAmp=$peakSessionAmplitude).")
    }

    /**
     * Submits a chunk of 16kHz linear PCM samples received from the audio engine.
     */
    fun submitPcmChunk(pcm16kHzSamples: ShortArray) {
        if (!isStopping.get()) {
            pcmChannel?.trySend(pcm16kHzSamples)
        }
    }

    /**

    suspend fun stopSession(isPickedUp: Boolean, saveToRepository: Boolean = true): VoicemailEntity? = withContext(Dispatchers.IO) {

                    if (finalResult.text.isNotBlank()) {
                        if (accumulatedSentences.isNotEmpty()) {
                            accumulatedSentences.append(" ")
                        }
                        accumulatedSentences.append(finalResult.text.trim())
                    }
                }
                s.release()
            }
        } catch (t: Throwable) {
            Log.w(TAG, "Error releasing Sherpa-ONNX stream", t)
        } finally {
            activeStream = null
        }

            tempPcmOutputStream?.close()
        } catch (ignored: Throwable) {
        } finally {
            tempPcmOutputStream = null
        }

        val pcm = tempPcmFile
        tempPcmFile = null

        if (!saveToRepository) {
            pcm?.delete()
            Log.i(TAG, "stopSession executed with saveToRepository=false (Self-Test mode). Temp files purged.")
            return@withContext null
        }

        val recordingsDir = File(context.filesDir, RECORDINGS_DIR).apply { mkdirs() }
        val cleanNumber = currentPhoneNumber.replace(Regex("[^0-9+]"), "").ifEmpty { "unknown" }
        val timeStamp = SimpleDateFormat("yyyyMMdd_HHmmss", Locale.US).format(Date(sessionStartTimeMs))
        val finalWavFile = File(recordingsDir, "voicemail_${cleanNumber}_$timeStamp.wav")

        val pcmLength = pcm?.length() ?: 0L
        if (pcm != null && pcm.exists() && pcmLength > 0L) {
            try {
                FileOutputStream(finalWavFile).use { fos ->
                    GreetingGenerator.writeWavHeader(
                        out = fos,
                        totalAudioLen = pcmLength,
                        sampleRate = SAMPLE_RATE_HZ,
                        channels = 1,
                        bitsPerSample = 16
                    )
                    FileInputStream(pcm).use { fis ->
                        fis.copyTo(fos)
                    }
                    fos.flush()
                }
                pcm.delete()
                Log.i(TAG, "Produced final WAV: ${finalWavFile.absolutePath} (${finalWavFile.length()} bytes, peakAmp=$peakSessionAmplitude)")
            } catch (e: Exception) {
                Log.e(TAG, "Failed creating WAV container from raw PCM file", e)
            }
        } else {
            try {
                val silenceBytes = ByteArray(SAMPLE_RATE_HZ * 2)
                FileOutputStream(finalWavFile).use { fos ->
                    GreetingGenerator.writeWavHeader(
                        out = fos,
                        totalAudioLen = silenceBytes.size.toLong(),
                        sampleRate = SAMPLE_RATE_HZ,
                        channels = 1,
                        bitsPerSample = 16
                    )
                    fos.write(silenceBytes)
                    fos.flush()
                }
                pcm?.delete()
                Log.i(TAG, "Produced baseline silence WAV: ${finalWavFile.absolutePath}")
            } catch (e: Exception) {
                Log.e(TAG, "Failed creating fallback WAV container", e)
            }
        }

        val durationSec = kotlin.math.max(1, (pcmLength / (SAMPLE_RATE_HZ * 2L)).toInt())
        val finalTranscript = accumulatedSentences.toString().trim()
            .ifEmpty {
                if (isPickedUp) ""
                else if (peakSessionAmplitude <= 10) "(Caller did not leave a spoken message)"
                else "(Voice message recorded â€” tap Play to listen)"
            }

        val entity = VoicemailEntity(
            callerName = currentCallerName,
            phoneNumber = currentPhoneNumber,
            simSlotIndex = currentSimSlotIndex,
            simDisplayName = currentSimDisplayName,
            timestamp = sessionStartTimeMs,
            durationSeconds = durationSec,
            transcript = finalTranscript,
            audioFilePath = finalWavFile.absolutePath,
            isRead = false,
            isPickedUp = isPickedUp
        )

        val id = repository.addVoicemail(entity)
        val persisted = entity.copy(id = id)

        _liveTranscript.value = finalTranscript
        _audioRmsLevel.value = 0f

        Log.i(TAG, "Voicemail recording saved successfully (ID: $id, Duration: ${durationSec}s, Caller: $currentPhoneNumber)")
        persisted
    }

    private fun calculateRms(samples: ShortArray): Float {
        if (samples.isEmpty()) return 0f
        var sumSquares = 0.0
        for (sample in samples) {
            val normalized = sample / 32768.0
            sumSquares += normalized * normalized
        }
        val meanSquare = sumSquares / samples.size
        return sqrt(meanSquare).toFloat().coerceIn(0f, 1f)
    }

    companion object {
        private const val TAG = "BootCompletedReceiver"
    }

    override fun onReceive(context: Context?, intent: Intent?) {

        val action = intent.action
        Log.i(TAG, "BootCompletedReceiver triggered with action: $action")

        if (action == Intent.ACTION_BOOT_COMPLETED ||
            action == Intent.ACTION_LOCKED_BOOT_COMPLETED ||
            action == Intent.ACTION_MY_PACKAGE_REPLACED ||
            action == "android.intent.action.QUICKBOOT_POWERON"

                val userManager = context.getSystemService(UserManager::class.java)
                if (userManager != null && !userManager.isUserUnlocked) {

                } else {
                    context
                }

                context
            }

            CoroutineScope(Dispatchers.IO).launch {

            synchronized(pendingConnectCallbacks) {
                pendingConnectCallbacks.add(onConnected)
            }
        }

            .debuggable(BuildConfig.DEBUG)
            .version(BuildConfig.VERSION_CODE)

            isConnecting.set(false)
        }
    }

                }

                if (!_isServiceBound.value) bindUserService()

        }

        Log.i(TAG, "onCallAdded: number=$displayNumber, caller=$callerName, SIM=$simLabel (slot=$simSlot), isOutgoing=$isOutgoing")

        val initialState = when {
            call.state == Call.STATE_RINGING -> TelecomCallManager.CallScreeningState.INCOMING_RINGING
            isOutgoing -> TelecomCallManager.CallScreeningState.OUTGOING_DIALING
            else -> TelecomCallManager.CallScreeningState.ACTIVE_CALL
        }

        TelecomCallManager.updateCall(
            call = call,
            name = callerName,
            number = displayNumber,
            simName = simLabel,
            simSlot = simSlot,
            state = initialState
        )

        updateInCallNotification()

        try {
            val activityIntent = Intent(this, InCallActivity::class.java).apply {
                addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_SINGLE_TOP or Intent.FLAG_ACTIVITY_CLEAR_TOP)
            }
            startActivity(activityIntent)
        } catch (t: Throwable) {
            Log.e(TAG, "Failed launching InCallActivity from onCallAdded", t)
        }

        if (call.state == Call.STATE_RINGING && !isOutgoing) {
            startAutoAnswerTimer()
        }
    }

    override fun onCallRemoved(call: Call) {
        super.onCallRemoved(call)
        call.unregisterCallback(callCallback)
        allCalls.remove(call)
        TelecomCallManager.updateCallsList(allCalls)
        if (allCalls.isEmpty()) {
            handleCallDisconnected()
        } else if (activeCall == call) {
            switchToNextActiveCall()
        }
    }

    private fun switchToNextActiveCall() {
        activeCall = allCalls.firstOrNull { it.state == Call.STATE_ACTIVE } ?: allCalls.firstOrNull()
        val current = activeCall ?: return
        val rawNumber = current.details?.handle?.schemeSpecificPart?.let { Uri.decode(it) }.orEmpty()
        val displayNumber = if (rawNumber.isNotBlank()) rawNumber else "Unknown Number"
        val callerName = current.details?.callerDisplayName?.takeIf { it.isNotBlank() }
            ?: resolveContactName(displayNumber)
        val (simLabel, simSlot) = resolveSimDetails(current.details?.accountHandle)
        val isOnHold = current.state == Call.STATE_HOLDING
        TelecomCallManager.setIsOnHold(isOnHold)
        TelecomCallManager.updateCall(
            call = current,
            name = callerName,
            number = displayNumber,
            simName = simLabel,
            simSlot = simSlot,
            state = TelecomCallManager.CallScreeningState.ACTIVE_CALL
        )
        updateInCallNotification()
    }

    override fun onCallAudioStateChanged(audioState: CallAudioState?) {
        super.onCallAudioStateChanged(audioState)
        if (isVoicemailScreeningActive && audioState != null) {
            val currentState = TelecomCallManager.screeningState.value
            val hasDigital = hasDigitalCapturePrivilege

            // Uplink muting enforcement:
            // During Phase B (SCREENING_LIVE_VOICEMAIL):
            // - If digital capture is active: keep Telecom uplink muted so caller only speaks.
            // - If unprivileged acoustic fallback: NEVER mute Telecom uplink because AudioFlinger zeroes AudioRecord!
            if (currentState == TelecomCallManager.CallScreeningState.SCREENING_LIVE_VOICEMAIL) {
                if (hasDigital) {
                    if (!audioState.isMuted) {
                        setMuted(true)
                    }
                } else {
                    if (audioState.isMuted) {
                        setMuted(false)
                    }
                }
            } else if (currentState == TelecomCallManager.CallScreeningState.SCREENING_GREETING ||
                currentState == TelecomCallManager.CallScreeningState.SCREENING_ANSWERING
            ) {
                // Phase A: Telecom uplink MUST remain unmuted so caller hears greeting!
                if (audioState.isMuted) {
                    setMuted(false)
                }
            }

            // Never switch to Loudspeaker unless the user explicitly enabled "Listen Live"!
            if (!TelecomCallManager.isScreeningSpeakerOn.value && (audioState.route and CallAudioState.ROUTE_SPEAKER) != 0) {
                setAudioRoute(CallAudioState.ROUTE_EARPIECE)
                audioManager.isSpeakerphoneOn = false
            }
        }
        TelecomCallManager.updateAudioState(audioState)
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_TRIGGER_VOICEMAIL -> triggerLiveVoicemailScreening(activeCall)
            ACTION_ANSWER_CALL -> TelecomCallManager.answerCallNormally()
            ACTION_PICK_UP -> takeOverVoicemailCall()
            ACTION_HANG_UP -> activeCall?.disconnect()
        }
        return super.onStartCommand(intent, flags, startId)
    }

    fun placeSecondCall(number: String) {
        try {
            val uri = Uri.fromParts("tel", number, null)
            val tm = getSystemService(Context.TELECOM_SERVICE) as? TelecomManager
            tm?.placeCall(uri, null)
        } catch (t: Throwable) {
            Log.e(TAG, "Error placing second call: ${t.message}")
        }
    }

    private fun startAutoAnswerTimer() {
        cancelAutoAnswerTimer()
        val runnable = Runnable {
            val call = activeCall
            if (call != null && call.state == Call.STATE_RINGING) {
                Log.i(TAG, "Auto-answer 8s timer fired. Initiating silent Live Voicemail.")
                triggerLiveVoicemailScreening(call)
            }
        }
        autoAnswerRunnable = runnable
        mainHandler.postDelayed(runnable, AUTO_ANSWER_DELAY_MS)
    }

    private fun cancelAutoAnswerTimer() {
        autoAnswerRunnable?.let { mainHandler.removeCallbacks(it) }
        autoAnswerRunnable = null
    }

    /**
     * Answers the call silently for Live Voicemail and launches the greeting + downlink capture pipeline.
     */
    fun triggerLiveVoicemailScreening(call: Call? = activeCall) {
        val targetCall = call ?: return
        cancelAutoAnswerTimer()

        if (isScreeningSessionRunning.getAndSet(true)) {
            Log.d(TAG, "Screening session already running for current call.")
            return
        }

        serviceScope.launch(Dispatchers.Main) {
            Log.i(TAG, "Triggering Live Voicemail. hasDigitalPrivilege=$hasDigitalCapturePrivilege")
            isVoicemailScreeningActive = true
            TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.SCREENING_ANSWERING)
            TelecomCallManager.setScreeningSpeakerStateOnly(false)
            updateInCallNotification()

            // Enforce Phase A audio state BEFORE answering call
            enforceSilentScreeningAudioState()

            if (targetCall.state == Call.STATE_RINGING || targetCall.state == Call.STATE_CONNECTING) {
                targetCall.answer(VideoProfile.STATE_AUDIO_ONLY)
            }

            var waitedMs = 0
            while (targetCall.state != Call.STATE_ACTIVE && targetCall.state != Call.STATE_DISCONNECTED && waitedMs < 5000) {
                delay(100)
                waitedMs += 100
            }

            if (targetCall.state == Call.STATE_DISCONNECTED) {
                Log.w(TAG, "Call disconnected before reaching STATE_ACTIVE")
                handleCallDisconnected()
                return@launch
            }

            // Re-enforce Phase A audio state as soon as Telecom activates call audio path
            enforceSilentScreeningAudioState()

            // Wait 600ms for VoLTE / 5G IMS RTP audio bearer setup so greeting is not dropped
            delay(600)
            enforceSilentScreeningAudioState()

            runCatching {
                setMuted(false)
                audioManager.isMicrophoneMute = false
            }

            executeVoicemailPipeline()
        }
    }

    /**
     * Enforces Phase A screening mode:
     * 1. Routes call audio to EARPIECE (never loudspeaker).
     * 2. Keeps Telecom uplink UNMUTED (`setMuted(false)`) and physical mic unmuted so greeting reaches caller.
     * 3. When digital privilege is held, mutes STREAM_VOICE_CALL so the receiver's phone is 100% dead silent.
     * 4. When unprivileged, sets STREAM_VOICE_CALL to earpiece level (70%).
     */
    private fun enforceSilentScreeningAudioState() {
        try {
            setAudioRoute(CallAudioState.ROUTE_EARPIECE)
            audioManager.isSpeakerphoneOn = false

            // Keep Telecom uplink UNMUTED during Phase A so greeting audio reaches caller
            setMuted(false)
            audioManager.isMicrophoneMute = false

            val hasDigital = hasDigitalCapturePrivilege
            if (hasDigital) {
                // With digital telephony privilege (UID 2000 or self-granted CAPTURE_AUDIO_OUTPUT + MODIFY_AUDIO_ROUTING),
                // MUTE STREAM_VOICE_CALL so room is 100% dead silent!
                audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_MUTE, 0)
            } else {
                // Without digital privilege, earpiece level (70% volume, strictly on earpiece)
                audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_UNMUTE, 0)
                val maxVoiceVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
                audioManager.setStreamVolume(AudioManager.STREAM_VOICE_CALL, (maxVoiceVol * 0.70f).toInt().coerceAtLeast(3), 0)
            }

            Log.i(TAG, "enforceSilentScreeningAudioState: ROUTE_EARPIECE, speaker=false, muted=false, hasDigital=$hasDigital")
        } catch (t: Throwable) {
            Log.w(TAG, "Failed enforcing silent screening audio state: ${t.message}")
        }
    }

    private suspend fun executeVoicemailPipeline() {
        isVoicemailScreeningActive = true
        isGreetingFinished.set(false)
        chunksReceivedCount.set(0)

        TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.SCREENING_GREETING)
        updateInCallNotification()

        val number = TelecomCallManager.phoneNumber.value
        val name = TelecomCallManager.callerName.value
        val simName = TelecomCallManager.simDisplayName.value
        val simSlot = TelecomCallManager.simSlotIndex.value

        transcriber.startSession(
            callerName = name,
            phoneNumber = number,
            simDisplayName = simName,
            simSlotIndex = simSlot
        )

        // Attempt background permanent permission self-granting if Shizuku / root is present
        withContext(Dispatchers.IO) {
            shizukuAudioController.grantPermanentStandalonePermissions()
        }

        val shizukuReady = withContext(Dispatchers.IO) {
            shizukuAudioController.updateShizukuStatus() && shizukuAudioController.ensureBound(2500L)
        }

        if (shizukuReady) {
            Log.i(TAG, "Privileged daemon bound! Dispatching privileged uplink greeting + downlink capture.")
            try {
                val afd = GreetingGenerator.getGreetingAssetFileDescriptor(this)
                shizukuAudioController.startVoicemailSession(afd, audioChunkCallback)

                greetingTimeoutJob?.cancel()
                greetingTimeoutJob = serviceScope.launch {
                    delay(8000L)
                    if (!isGreetingFinished.get() && isVoicemailScreeningActive) {
                        Log.w(TAG, "Privileged greeting timeout (8s) reached; ensuring SCREENING_LIVE_VOICEMAIL state.")
                        onGreetingPlaybackFinished()
                    }
                }
                return
            } catch (t: Throwable) {
                Log.e(TAG, "Failed starting privileged voicemail session, switching to standalone engine", t)
            }
        }

        Log.i(TAG, "Using StandaloneTelephonyAudioEngine for greeting injection & downlink capture.")
        standaloneEngine.playGreeting(this@VoicemailInCallService) {
            mainHandler.post {
                onGreetingPlaybackFinished()
                standaloneEngine.startCapture(this@VoicemailInCallService) { chunk ->
                    if (isVoicemailScreeningActive && isGreetingFinished.get()) {
                        transcriber.submitPcmChunk(chunk)
                    }
                }
            }
        }
    }

    private fun onGreetingPlaybackFinished() {
        if (isGreetingFinished.getAndSet(true)) return
        greetingTimeoutJob?.cancel()
        greetingTimeoutJob = null

        if (!isVoicemailScreeningActive) return

        val hasDigital = hasDigitalCapturePrivilege
        Log.i(TAG, "Greeting finished. Transitioning to SCREENING_LIVE_VOICEMAIL (hasDigitalCapture=$hasDigital).")

        try {
            audioManager.setParameters("incall_music_enabled=false;incall_music_uplink=false")
            audioManager.setParameters("Set_SpeechCall_Music_Uplink=0")
            audioManager.setParameters("g_call_sim_voice_uplink=false")

            if (hasDigital) {
                // We can safely mute the physical mic because VOICE_DOWNLINK (audio source 3)
                // captures the cellular RX leg before the mic mixer!
                setMuted(true)
                audioManager.isMicrophoneMute = true
            } else {
                // Unprivileged acoustic fallback:
                // NEVER call setMuted(true) or audioManager.isMicrophoneMute = true during Phase B!
                // If you call setMuted(true), AudioRecord is muted by the OS and records 0 bytes of audio!
                setMuted(false)
                audioManager.isMicrophoneMute = false
                setAudioRoute(CallAudioState.ROUTE_EARPIECE)
                audioManager.isSpeakerphoneOn = false
                val maxVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
                audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_UNMUTE, 0)
                audioManager.setStreamVolume(AudioManager.STREAM_VOICE_CALL, (maxVol * 0.6f).toInt().coerceAtLeast(2), 0)
            }

            // Respect user's "Listen Live" toggle (default is FALSE = silent earpiece in room)
            val listenLive = TelecomCallManager.isScreeningSpeakerOn.value
            setScreeningSpeakerEnabled(listenLive)
        } catch (t: Throwable) {
            Log.w(TAG, "Error configuring post-greeting audio state: ${t.message}")
        }

        TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.SCREENING_LIVE_VOICEMAIL)
        updateInCallNotification()
    }

    /**
     * Handles user tapping "Pick Up" to take over the call and talk to the caller live.
     */
    fun takeOverVoicemailCall() {
        Log.i(TAG, "Taking over Live Voicemail session: stopping screening and unmuting user microphone.")
        isVoicemailScreeningActive = false
        standaloneEngine.stopSession()
        shizukuAudioController.stopVoicemailSession()

        // 1. Unmute hardware microphone & restore normal earpiece/call volume for live conversation
        setMuted(false)
        audioManager.isMicrophoneMute = false
        audioManager.isSpeakerphoneOn = false
        setAudioRoute(CallAudioState.ROUTE_EARPIECE)
        audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_UNMUTE, 0)
        val maxVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
        audioManager.setStreamVolume(AudioManager.STREAM_VOICE_CALL, (maxVol * 0.8f).toInt().coerceAtLeast(3), 0)

        // 2. Acquire proximity lock so ear proximity blanks display
        acquireProximityWakeLock()

        // 3. Update UI state to ActiveCall
        TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.ACTIVE_CALL)
        updateInCallNotification()

        // 4. Finalize voicemail record as picked up
        stopAndSaveVoicemailSession(isPickedUp = true)
    }

    /**
     * Toggles whether the user wants to hear the caller's voice out loud in the room while screening.
     * When digital capture is active, microphone remains strictly muted (`setMuted(true)`).
     * When unprivileged acoustic fallback is active, mic must stay unmuted (`setMuted(false)`).
     */
    fun setScreeningSpeakerEnabled(enabled: Boolean) {
        try {
            val hasDigital = hasDigitalCapturePrivilege

            if (hasDigital) {
                setMuted(true)
                audioManager.isMicrophoneMute = true
            } else {
                setMuted(false)
                audioManager.isMicrophoneMute = false
            }

            shizukuAudioController.setScreeningSpeakerEnabled(enabled)

            if (enabled) {
                setAudioRoute(CallAudioState.ROUTE_SPEAKER)
                audioManager.isSpeakerphoneOn = true
                audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_UNMUTE, 0)
                val maxVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
                audioManager.setStreamVolume(AudioManager.STREAM_VOICE_CALL, (maxVol * 0.7f).toInt().coerceAtLeast(2), 0)
            } else {
                setAudioRoute(CallAudioState.ROUTE_EARPIECE)
                audioManager.isSpeakerphoneOn = false
                if (hasDigital) {
                    audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_MUTE, 0)
                } else {
                    // In unprivileged standalone mode: keep ROUTE_EARPIECE at 60% volume so caller audio is captured
                    audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_UNMUTE, 0)
                    val maxVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
                    audioManager.setStreamVolume(AudioManager.STREAM_VOICE_CALL, (maxVol * 0.6f).toInt().coerceAtLeast(2), 0)
                }
            }
            Log.i(TAG, "setScreeningSpeakerEnabled($enabled): hasDigital=$hasDigital, speaker=$enabled")
        } catch (t: Throwable) {
            Log.e(TAG, "Error toggling screening speaker: ${t.message}")
        }
    }

    fun stopAndSaveVoicemailSession(isPickedUp: Boolean = false, onComplete: (() -> Unit)? = null) {
        if (hasSavedCurrentSession.getAndSet(true)) {
            onComplete?.invoke()
            return
        }

        cancelAutoAnswerTimer()
        releaseProximityWakeLock()
        standaloneEngine.stopSession()
        shizukuAudioController.stopVoicemailSession()

        val wasScreening = isVoicemailScreeningActive || isScreeningSessionRunning.get()
        val wasGreetingPlaying = !isGreetingFinished.get()

        isVoicemailScreeningActive = false
        isScreeningSessionRunning.set(false)

        // Restore normal audio state when session ends
        setMuted(false)
        audioManager.isMicrophoneMute = false
        audioManager.isSpeakerphoneOn = false
        audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_UNMUTE, 0)

        TelecomCallManager.setScreeningState(
            if (isPickedUp) TelecomCallManager.CallScreeningState.ACTIVE_CALL
            else TelecomCallManager.CallScreeningState.DISCONNECTED
        )

        appScope.launch(NonCancellable) {
            try {
                if (wasScreening) {
                    val voicemail = transcriber.stopSession(isPickedUp = isPickedUp, saveToRepository = true)
                    if (voicemail != null) {
                        if (wasGreetingPlaying && !isPickedUp) {
                            VoicemailApplication.voicemailRepository.updateTranscript(voicemail.id, "Caller hung up during greeting")
                            postVoicemailCompletedNotification(voicemail.callerName ?: voicemail.phoneNumber, "Caller hung up during greeting")
                        } else if (!isPickedUp && voicemail.transcript.isNotBlank()) {
                            postVoicemailCompletedNotification(voicemail.callerName ?: voicemail.phoneNumber, voicemail.transcript)
                        }
                    }
                }
            } catch (t: Throwable) {
                Log.e(TAG, "Error in stopSession / save", t)
            } finally {
                withContext(Dispatchers.Main) {
                    onComplete?.invoke()
                }
            }
        }
    }

    private fun handleCallDisconnected() {
        cancelAutoAnswerTimer()
        releaseProximityWakeLock()
        stopAndSaveVoicemailSession(isPickedUp = false) {
            try {
                stopForeground(STOP_FOREGROUND_REMOVE)
                val nm = getSystemService(NotificationManager::class.java)
                nm.cancel(INCOMING_NOTIFICATION_ID)
            } catch (t: Throwable) {
                Log.w(TAG, "Error dismissing in-call notification: ${t.message}")
            }
        }

        mainHandler.postDelayed({
            if (allCalls.isEmpty()) {
                TelecomCallManager.reset()
                activeCall = null
            }
        }, 1500L)
    }

    private fun updateInCallNotification() {
        val name = TelecomCallManager.callerName.value ?: TelecomCallManager.phoneNumber.value.ifBlank { "Phone Call" }
        val simLabel = TelecomCallManager.simDisplayName.value
        val state = TelecomCallManager.screeningState.value

        val (titleText, contentText) = when (state) {
            TelecomCallManager.CallScreeningState.IDLE ->
                "Live Voicemail" to "Ready"
            TelecomCallManager.CallScreeningState.INCOMING_RINGING ->
                "Incoming Call â€¢ $simLabel" to "Ringing from $name"
            TelecomCallManager.CallScreeningState.SCREENING_ANSWERING ->
                "Answering Silently..." to "Connecting Live Voicemail..."
            TelecomCallManager.CallScreeningState.SCREENING_GREETING ->
                "Screening Call â€¢ $simLabel" to "Playing greeting to $name..."
            TelecomCallManager.CallScreeningState.SCREENING_LIVE_VOICEMAIL ->
                "Live Voicemail â€¢ $simLabel" to "Recording voice message from $name"
            TelecomCallManager.CallScreeningState.ACTIVE_CALL ->
                "Call in Progress â€¢ $simLabel" to "Talking with $name"
            TelecomCallManager.CallScreeningState.OUTGOING_DIALING ->
                "Outgoing Call â€¢ $simLabel" to "Calling $name..."
            TelecomCallManager.CallScreeningState.DISCONNECTED ->
                "Call Ended" to "Disconnected"
        }

        val activityIntent = Intent(this, InCallActivity::class.java).apply {
            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_SINGLE_TOP)
        }
        val contentPendingIntent = android.app.PendingIntent.getActivity(
            this,
            0,
            activityIntent,
            android.app.PendingIntent.FLAG_UPDATE_CURRENT or android.app.PendingIntent.FLAG_IMMUTABLE
        )

        val builder = NotificationCompat.Builder(this, VoicemailApplication.CHANNEL_INCALL_ID)
            .setSmallIcon(android.R.drawable.stat_sys_phone_call)
            .setContentTitle(titleText)
            .setContentText(contentText)
            .setPriority(NotificationCompat.PRIORITY_MAX)
            .setCategory(NotificationCompat.CATEGORY_CALL)
            .setContentIntent(contentPendingIntent)
            .setOngoing(true)

        if (state == TelecomCallManager.CallScreeningState.SCREENING_LIVE_VOICEMAIL ||
            state == TelecomCallManager.CallScreeningState.SCREENING_GREETING
        ) {
            val pickUpIntent = Intent(this, VoicemailInCallService::class.java).apply {
                action = ACTION_PICK_UP
            }
            val pickUpPendingIntent = android.app.PendingIntent.getService(
                this,
                1,
                pickUpIntent,
                android.app.PendingIntent.FLAG_UPDATE_CURRENT or android.app.PendingIntent.FLAG_IMMUTABLE
            )
            builder.addAction(android.R.drawable.sym_call_incoming, "Pick Up", pickUpPendingIntent)

            val hangUpIntent = Intent(this, VoicemailInCallService::class.java).apply {
                action = ACTION_HANG_UP
            }
            val hangUpPendingIntent = android.app.PendingIntent.getService(
                this,
                2,
                hangUpIntent,
                android.app.PendingIntent.FLAG_UPDATE_CURRENT or android.app.PendingIntent.FLAG_IMMUTABLE
            )
            builder.addAction(android.R.drawable.ic_menu_close_clear_cancel, "End Call", hangUpPendingIntent)
        }

        try {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                startForeground(
                    INCOMING_NOTIFICATION_ID,
                    builder.build(),
                    android.content.pm.ServiceInfo.FOREGROUND_SERVICE_TYPE_PHONE_CALL
                )
            } else {
                startForeground(INCOMING_NOTIFICATION_ID, builder.build())
            }
        } catch (t: Throwable) {
            Log.w(TAG, "Error displaying foreground notification: ${t.message}")
        }
    }

    private fun postVoicemailCompletedNotification(caller: String, transcript: String) {
        val intent = Intent(this, com.personal.voicemail.ui.MainActivity::class.java)
        val pendingIntent = android.app.PendingIntent.getActivity(
            this,
            0,
            intent,
            android.app.PendingIntent.FLAG_UPDATE_CURRENT or android.app.PendingIntent.FLAG_IMMUTABLE
        )

        val notif = NotificationCompat.Builder(this, VoicemailApplication.CHANNEL_VOICEMAIL_NOTIFICATIONS_ID)
            .setSmallIcon(android.R.drawable.sym_call_incoming)
            .setContentTitle("New Voicemail from $caller")
            .setContentText(transcript.ifBlank { "New voice message received" })
            .setStyle(NotificationCompat.BigTextStyle().bigText(transcript))
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setAutoCancel(true)
            .setContentIntent(pendingIntent)
            .build()

        val nm = getSystemService(NotificationManager::class.java)
        nm.notify(System.currentTimeMillis().toInt(), notif)
    }

    private fun acquireProximityWakeLock() {
        if (proximityWakeLock == null) {
            val pm = getSystemService(Context.POWER_SERVICE) as? PowerManager
            if (pm != null && pm.isWakeLockLevelSupported(PowerManager.PROXIMITY_SCREEN_OFF_WAKE_LOCK)) {
                proximityWakeLock = pm.newWakeLock(
                    PowerManager.PROXIMITY_SCREEN_OFF_WAKE_LOCK,
                    "VoicemailApp:InCallProximityWakeLock"
                )
            }
        }
        proximityWakeLock?.let {
            if (!it.isHeld) {
                it.acquire(60 * 60 * 1000L) // 1 hour safety timeout
            }
        }
    }

    private fun releaseProximityWakeLock() {
        proximityWakeLock?.let {
            if (it.isHeld) {
                it.release()
            }
        }
        proximityWakeLock = null
    }

    private fun resolveContactName(phoneNumber: String): String? {
        if (phoneNumber.isBlank()) return null
        return try {
            val uri = Uri.withAppendedPath(
                ContactsContract.PhoneLookup.CONTENT_FILTER_URI,
                Uri.encode(phoneNumber)
            )
            val projection = arrayOf(ContactsContract.PhoneLookup.DISPLAY_NAME)
            contentResolver.query(uri, projection, null, null, null)?.use { cursor ->
                if (cursor.moveToFirst()) {
                    cursor.getString(0)
                } else null
            }
        } catch (t: Throwable) {
            null
        }
    }

    private fun resolveSimDetails(handle: PhoneAccountHandle?): Pair<String, Int> {
        if (handle == null) return "SIM 1" to 0
        return try {
            val sm = getSystemService(Context.TELEPHONY_SUBSCRIPTION_SERVICE) as? SubscriptionManager
            val subList = sm?.activeSubscriptionInfoList.orEmpty()
            val matchedInfo = subList.firstOrNull { it.iccId == handle.id || it.subscriptionId.toString() == handle.id }
                ?: subList.getOrNull(0)

            val displayName = matchedInfo?.displayName?.toString()

private fun querySimAccounts(context: Context): List<SimAccountOption> {
    val results = ArrayList<SimAccountOption>()
    val telecomManager = context.getSystemService(TelecomManager::class.java)
    val subscriptionManager = context.getSystemService(SubscriptionManager::class.java)

    if (ContextCompat.checkSelfPermission(context, android.Manifest.permission.READ_PHONE_STATE) != PackageManager.PERMISSION_GRANTED) {
        return listOf(SimAccountOption(0, "SIM 1", null))
    }

    try {
        val activeSubs = subscriptionManager?.activeSubscriptionInfoList ?: emptyList<SubscriptionInfo>()
        val callHandles = telecomManager?.callCapablePhoneAccounts ?: emptyList<PhoneAccountHandle>()

        if (activeSubs.isNotEmpty()) {
            for (sub in activeSubs) {
                val matchedHandle = callHandles.firstOrNull { it.id == sub.subscriptionId.toString() || it.id == sub.iccId }
                val name = sub.displayName?.toString()?.ifBlank { sub.carrierName?.toString() } ?: "SIM ${sub.simSlotIndex + 1}"
                results.add(
                    SimAccountOption(
                        slotIndex = sub.simSlotIndex,
                        displayName = name,
                        phoneAccountHandle = matchedHandle
                    )
                )
            }
        } else {
            results.add(SimAccountOption(0, "SIM 1", callHandles.firstOrNull()))
        }
    } catch (e: Exception) {
        results.add(SimAccountOption(0, "SIM 1", null))
    }

    return results.sortedBy { it.slotIndex }
}

private fun placeOutgoingCall(
    context: Context,
    number: String,
    accountHandle: PhoneAccountHandle?
) {
    val telecomManager = context.getSystemService(TelecomManager::class.java) ?: return
    val uri = Uri.fromParts("tel", number, null)
    val extras = Bundle().apply {
        if (accountHandle != null) {
            putParcelable(TelecomManager.EXTRA_PHONE_ACCOUNT_HANDLE, accountHandle)
        }
    }

    try {
        if (ContextCompat.checkSelfPermission(context, android.Manifest.permission.CALL_PHONE) == PackageManager.PERMISSION_GRANTED) {
            telecomManager.placeCall(uri, extras)
        }
    } catch (e: Exception) {
        android.util.Log.e("DialpadTab", "Failed to place call to $number", e)
    }
}

private suspend fun fetchCallLogs(context: Context): List<CallLogEntry> = withContext(Dispatchers.IO) {
    if (ContextCompat.checkSelfPermission(context, android.Manifest.permission.READ_CALL_LOG) != PackageManager.PERMISSION_GRANTED) {
        return@withContext emptyList()
    }

    val result = ArrayList<CallLogEntry>()
    val projection = arrayOf(
        CallLog.Calls._ID,
        CallLog.Calls.NUMBER,
        CallLog.Calls.CACHED_NAME,
        CallLog.Calls.DATE,
        CallLog.Calls.DURATION,
        CallLog.Calls.TYPE,
        CallLog.Calls.PHONE_ACCOUNT_ID
    )

    try {
        context.contentResolver.query(
            CallLog.Calls.CONTENT_URI,
            projection,
            null,
            null,
            "${CallLog.Calls.DATE} DESC LIMIT 100"
        )?.use { cursor ->
            val colId = cursor.getColumnIndex(CallLog.Calls._ID)
            val colNumber = cursor.getColumnIndex(CallLog.Calls.NUMBER)
            val colName = cursor.getColumnIndex(CallLog.Calls.CACHED_NAME)
            val colDate = cursor.getColumnIndex(CallLog.Calls.DATE)
            val colDuration = cursor.getColumnIndex(CallLog.Calls.DURATION)
            val colType = cursor.getColumnIndex(CallLog.Calls.TYPE)
            val colSimId = cursor.getColumnIndex(CallLog.Calls.PHONE_ACCOUNT_ID)

            while (cursor.moveToNext()) {
                val id = cursor.getLong(colId)
                val number = cursor.getString(colNumber) ?: "Unknown"
                val name = cursor.getString(colName)
                val date = cursor.getLong(colDate)
                val duration = cursor.getLong(colDuration)
                val type = cursor.getInt(colType)
                val simId = if (colSimId != -1) cursor.getString(colSimId) else null

                result.add(
                    CallLogEntry(
                        id = id,
                        number = number,
                        name = name,
                        date = date,
                        durationSec = duration,
                        type = type,
                        simAccountId = simId
                    )
                )
            }
        }
    } catch (e: Exception) {
        android.util.Log.e("RecentsTab", "Error reading call logs", e)
    }

    result
}

private fun formatLogDate(dateMs: Long): String {
    val date = Date(dateMs)
    val now = Date()
    val cal1 = java.util.Calendar.getInstance().apply { time = date }
    val cal2 = java.util.Calendar.getInstance().apply { time = now }

    return if (cal1.get(java.util.Calendar.YEAR) == cal2.get(java.util.Calendar.YEAR) &&
        cal1.get(java.util.Calendar.DAY_OF_YEAR) == cal2.get(java.util.Calendar.DAY_OF_YEAR)
    ) {
        SimpleDateFormat("h:mm a", Locale.getDefault()).format(date)
    } else {
        val diffDays = (now.time - dateMs) / (1000 * 60 * 60 * 24)
        if (diffDays < 7) {
            SimpleDateFormat("EEEE", Locale.getDefault()).format(date)
        } else {
            SimpleDateFormat("MM/dd/yy", Locale.getDefault()).format(date)
        }
    }
}

/**
 * Strict, accurate battery optimization exemption check:
 * 1. Checks live PowerManager.isIgnoringBatteryOptimizations
 * 2. Only during the first 120s after boot, falls back to previously confirmed state
 *    while PowerManager initializes.
 * 3. Does NOT check AppOps OP_RUN_ANY_IN_BACKGROUND (which is allowed by default and gives false positives).
 */
fun checkBatteryOptimizationExempt(context: Context): Boolean {
    val powerManager = context.getSystemService(PowerManager::class.java)
    if (powerManager?.isIgnoringBatteryOptimizations(context.packageName) == true) {
        SetupPreferences.setBatteryExemptConfirmed(context, true)
        return true
    }

    val isRecentBoot = SystemClock.elapsedRealtime() < 120_000L
    if (isRecentBoot && SetupPreferences.isBatteryExemptConfirmed(context)) {
        return true
    }

    SetupPreferences.setBatteryExemptConfirmed(context, false)
    return false
}

/**
 * Multi-layer check for Accessibility Service:
 * 1. Live bound instance check ([VoicemailAccessibilityService.instance])
 * 2. AccessibilityManager.getEnabledAccessibilityServiceList
 * 3. Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES unflattened component parsing
 * 4. Boot-grace period (first 60s) fallback if previously verified.
 */
fun checkAccessibilityServiceEnabled(context: Context): Boolean {
    // 1. Live instance bound in RAM
    if (VoicemailAccessibilityService.instance != null) {
        SetupPreferences.setAccessibilityConfirmed(context, true)
        return true
    }

    // 2. Query AccessibilityManager for enabled services
    try {
        val am = context.getSystemService(Context.ACCESSIBILITY_SERVICE) as? AccessibilityManager
        if (am != null) {
            val enabledList = am.getEnabledAccessibilityServiceList(android.accessibilityservice.AccessibilityServiceInfo.FEEDBACK_ALL_MASK)
            val isPresent = enabledList.any { info ->
                val serviceInfo = info.resolveInfo?.serviceInfo
                serviceInfo?.packageName.equals(context.packageName, ignoreCase = true) &&
                        serviceInfo?.name?.contains("VoicemailAccessibilityService", ignoreCase = true) == true
            }
            if (isPresent) {
                SetupPreferences.setAccessibilityConfirmed(context, true)
                return true
            }
        }
    } catch (ignored: Throwable) {}

    // 3. Parse Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES
    try {
        val enabledServices = Settings.Secure.getString(
            context.contentResolver,
            Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES
        )
        if (!enabledServices.isNullOrBlank()) {
            val colonSplitter = TextUtils.SimpleStringSplitter(':')
            colonSplitter.setString(enabledServices)
            while (colonSplitter.hasNext()) {
                val componentNameString = colonSplitter.next()
                val enabledComponent = ComponentName.unflattenFromString(componentNameString)
                if (enabledComponent != null &&
                    enabledComponent.packageName.equals(context.packageName, ignoreCase = true) &&
                    enabledComponent.className.contains("VoicemailAccessibilityService", ignoreCase = true)
                ) {
                    SetupPreferences.setAccessibilityConfirmed(context, true)
                    return true
                }
            }
        }
    } catch (ignored: Throwable) {}

    // 4. Fallback only during the first 60 seconds after boot while system binds services
    val isRecentBoot = SystemClock.elapsedRealtime() < 60_000L
    if (isRecentBoot && SetupPreferences.isAccessibilityConfirmed(context)) {
        return true
    }

    SetupPreferences.setAccessibilityConfirmed(context, false)
    return false
}

        }
        context.startActivity(intent)
    } catch (t: Throwable) {
        android.util.Log.e("SystemReadiness", "Failed opening accessibility settings", t)
    }

    var isDraggingSlider by remember { mutableStateOf(false) }

    val mainHandler = remember { Handler(Looper.getMainLooper()) }

    DisposableEffect(audioPath, isPlaying) {
        if (isPlaying) {
            try {
                val mp = MediaPlayer().apply {
                    setDataSource(audioPath)
                    setOnErrorListener { _, what, extra ->
                        Log.e("VoicemailPlayer", "MediaPlayer playback error: what=$what extra=$extra")
                        onPlayToggle()
                        true
                    }
                    prepare()
                    totalDurationMs = duration
                    start()
                    setOnCompletionListener {
                        onPlayToggle()
                    }
                }
                mediaPlayer = mp

                val progressRunnable = object : Runnable {
                    override fun run() {
                        val active = mediaPlayer ?: return
                        val playing = runCatching { active.isPlaying }.getOrDefault(false)
                        if (playing && !isDraggingSlider) {
                            runCatching { currentPositionMs = active.currentPosition }
                            mainHandler.postDelayed(this, 100)
                        }
                    }
                }
                mainHandler.post(progressRunnable)
            } catch (e: Exception) {
                Log.e("VoicemailPlayer", "Error initializing MediaPlayer for $audioPath", e)
                onPlayToggle()
            }
        } else {
            mainHandler.removeCallbacksAndMessages(null)
            mediaPlayer?.let {
                runCatching {
                    if (it.isPlaying) it.stop()
                    it.release()
                }
            }
            mediaPlayer = null
            currentPositionMs = 0
        }

        onDispose {
            mainHandler.removeCallbacksAndMessages(null)
            mediaPlayer?.let {
                runCatching {
                    if (it.isPlaying) it.stop()
                    it.release()
                }
            }
            mediaPlayer = null
        }
    }

    Row(
        modifier = Modifier.fillMaxWidth(),
        verticalAlignment = Alignment.CenterVertically
    ) {
        // Play/Pause circular button
        Box(
            modifier = Modifier
                .size(40.dp)
                .clip(CircleShape)
                .background(if (isPlaying) AccentBlue else GlassSurface)
                .border(1.dp, GlassBorder, CircleShape)
                .clickable { onPlayToggle() },
            contentAlignment = Alignment.Center
        ) {
            Icon(
                imageVector = if (isPlaying) Icons.Default.Pause else Icons.Default.PlayArrow,
                contentDescription = if (isPlaying) "Pause" else "Play",
                tint = Color.White,
                modifier = Modifier.size(20.dp)
            )
        }

        Spacer(modifier = Modifier.width(12.dp))

        // Seekbar slider
        Column(modifier = Modifier.weight(1f)) {
            val progress = if (totalDurationMs > 0) {
                (currentPositionMs.toFloat() / totalDurationMs).coerceIn(0f, 1f)
            } else 0f

            Slider(
                value = progress,
                onValueChange = { newFrac ->
                    isDraggingSlider = true
                    currentPositionMs = (newFrac * totalDurationMs).toInt()
                },
                onValueChangeFinished = {
                    mediaPlayer?.seekTo(currentPositionMs)
                    isDraggingSlider = false
                },
                colors = SliderDefaults.colors(

                    activeTrackColor = AccentBlue,
                    inactiveTrackColor = Color(0x33FFFFFF)

private fun formatDate(epochMs: Long): String {
    val date = Date(epochMs)
    val now = Date()
    val cal1 = java.util.Calendar.getInstance().apply { time = date }
    val cal2 = java.util.Calendar.getInstance().apply { time = now }

    return if (cal1.get(java.util.Calendar.YEAR) == cal2.get(java.util.Calendar.YEAR) &&
        cal1.get(java.util.Calendar.DAY_OF_YEAR) == cal2.get(java.util.Calendar.DAY_OF_YEAR)
    ) {
        SimpleDateFormat("h:mm a", Locale.getDefault()).format(date)
    } else {
        SimpleDateFormat("MMM d, h:mm a", Locale.getDefault()).format(date)
    }
}

private fun formatDuration(seconds: Int): String {
    val m = seconds / 60
    val s = seconds % 60
    return String.format("%02d:%02d", m, s)
}

                    .fillMaxWidth()
                    .padding(vertical = 16.dp),
                contentAlignment = Alignment.Center
            ) {
                when (screeningState) {
                    CallScreeningState.SCREENING_ANSWERING,
                    CallScreeningState.SCREENING_GREETING,
                    CallScreeningState.SCREENING_LIVE_VOICEMAIL -> {
                        ScreeningTranscriptCard(
                            transcript = liveTranscript,
                            isGreeting = screeningState == CallScreeningState.SCREENING_GREETING || screeningState == CallScreeningState.SCREENING_ANSWERING,
                            rmsLevel = audioRms,
                            isSpeakerListening = isScreeningSpeakerOn
                        )
                    }
                    CallScreeningState.OUTGOING_DIALING -> {
                        OutgoingCallDialingCard()
                    }
                    CallScreeningState.ACTIVE_CALL -> {
                        ActiveCallWaveformCard(callDuration = callDuration, isHold = isHold)
                    }
                    else -> {
                        IncomingCallRingingCard(rmsLevel = audioRms)
                    }
                }
            }

            // Bottom Controls Section
            when (screeningState) {
                CallScreeningState.INCOMING_RINGING -> {
                    IncomingCallControls(
                        onVoicemail = { TelecomCallManager.sendToLiveVoicemail() },
                        onDecline = { TelecomCallManager.endCall() },
                        onAnswer = { TelecomCallManager.answerCallNormally() }
                    )
                }

                CallScreeningState.OUTGOING_DIALING -> {
                    OutgoingCallControls(
                        isMuted = isMuted,
                        isSpeakerOn = isSpeakerOn,
                        onToggleMute = { TelecomCallManager.toggleMute() },
                        onToggleKeypad = { showKeypadOverlay = !showKeypadOverlay },
                        onToggleSpeaker = { TelecomCallManager.toggleSpeaker() },
                        onEndCall = { TelecomCallManager.endCall() }
                    )
                }

                CallScreeningState.SCREENING_ANSWERING,
                CallScreeningState.SCREENING_GREETING,
                CallScreeningState.SCREENING_LIVE_VOICEMAIL -> {
                    ScreeningCallControls(
                        isSpeakerOn = isScreeningSpeakerOn,
                        onToggleSpeaker = {
                            TelecomCallManager.setScreeningSpeaker(!isScreeningSpeakerOn)
                        },
                        onPickUp = { TelecomCallManager.pickUpVoicemail() },
                        onEndCall = { TelecomCallManager.endCall() }
                    )
                }

                CallScreeningState.ACTIVE_CALL -> {
                    ActiveCallControls(
                        isMuted = isMuted,
                        isHold = isHold,
                        isSpeakerOn = isSpeakerOn,
                        canConference = canConference,
                        hasMultipleCalls = hasMultipleCalls,
                        onToggleMute = { TelecomCallManager.toggleMute() },
                        onToggleKeypad = { showKeypadOverlay = !showKeypadOverlay },
                        onToggleSpeaker = { TelecomCallManager.toggleSpeaker() },
                        onToggleHold = { TelecomCallManager.toggleHold() },
                        onAddCall = { showAddCallOverlay = true },
                        onMergeCalls = { TelecomCallManager.mergeConference() },
                        onSwapCalls = { TelecomCallManager.swapCalls() },
                        onSendToVoicemail = { TelecomCallManager.sendToLiveVoicemail() },
                        onEndCall = { TelecomCallManager.endCall() }
                    )
                }

                else -> {
                    Spacer(modifier = Modifier.height(80.dp))
                }
            }
        }

        // Interactive In-Call DTMF Keypad Overlay
        AnimatedVisibility(
            visible = showKeypadOverlay,
            enter = slideInVertically { it } + fadeIn(),
            exit = slideOutVertically { it } + fadeOut()
        ) {
            DtmfKeypadOverlay(
                onClose = { showKeypadOverlay = false },
                onKeyPress = { digit -> TelecomCallManager.playDtmfTone(digit) },
                onKeyRelease = { TelecomCallManager.stopDtmfTone() }
            )
        }

        // Add Call Overlay for placing a second call / conference
        AnimatedVisibility(
            visible = showAddCallOverlay,
            enter = slideInVertically { it } + fadeIn(),
            exit = slideOutVertically { it } + fadeOut()
        ) {
            AddCallOverlay(
                onClose = { showAddCallOverlay = false },
                onPlaceCall = { number ->
                    showAddCallOverlay = false
                    TelecomCallManager.currentInCallService?.placeSecondCall(number)
                }
            )
        }
    }
}

@Composable
private fun StatusPill(
    state: CallScreeningState,
    durationSeconds: Int,
    rmsLevel: Float,
    isHold: Boolean = false
) {
    val pillBg = when (state) {
        CallScreeningState.SCREENING_ANSWERING -> Color(0x33FF9500)
        CallScreeningState.SCREENING_GREETING -> Color(0x33FF9500)
        CallScreeningState.SCREENING_LIVE_VOICEMAIL -> Color(0x3334C759)
        CallScreeningState.ACTIVE_CALL -> if (isHold) Color(0x33FF9500) else Color(0x33007AFF)
        CallScreeningState.OUTGOING_DIALING -> Color(0x33007AFF)
        else -> GlassSurface
    }

    val pillText = when (state) {
        CallScreeningState.INCOMING_RINGING -> "Incoming Callâ€¦"
        CallScreeningState.OUTGOING_DIALING -> "Callingâ€¦"
        CallScreeningState.SCREENING_ANSWERING -> "Answering for Voicemailâ€¦"
        CallScreeningState.SCREENING_GREETING -> "Greeting Callerâ€¦"
        CallScreeningState.SCREENING_LIVE_VOICEMAIL -> "Live Voicemail"
        CallScreeningState.ACTIVE_CALL -> if (isHold) "On Hold" else formatCallDuration(durationSeconds)
        CallScreeningState.DISCONNECTED -> "Call Ended"
        CallScreeningState.IDLE -> ""
    }

    Row(
        modifier = Modifier
            .clip(RoundedCornerShape(20.dp))
            .background(pillBg)
            .border(1.dp, GlassBorder, RoundedCornerShape(20.dp))
            .padding(horizontal = 14.dp, vertical = 6.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        if (state == CallScreeningState.SCREENING_LIVE_VOICEMAIL) {
            LiveSpeechWaveform(
                rmsLevel = rmsLevel,
                isSpeaking = true,
                modifier = Modifier.padding(end = 8.dp)
            )
        }

        Text(
            text = pillText,
            fontSize = 13.sp,
            fontWeight = FontWeight.SemiBold,
            color = TextPrimary
        )
    }
}

@Composable
private fun ScreeningTranscriptCard(
    transcript: String,
    isGreeting: Boolean,
    rmsLevel: Float,
    isSpeakerListening: Boolean
) {
    val listState = rememberLazyListState()

    // Auto-scroll to bottom as new speech words arrive
    LaunchedEffect(transcript) {
        if (transcript.isNotEmpty()) {
            listState.animateScrollToItem(1)
        }
    }

    GlassmorphicCard(
        modifier = Modifier
            .fillMaxSize()
            .padding(vertical = 4.dp),
        cornerRadius = 28.dp,
        backgroundColor = Color(0x24FFFFFF)
    ) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(20.dp),
            verticalArrangement = Arrangement.SpaceBetween
        ) {
            // Live Screening Status Header
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Box(
                        modifier = Modifier
                            .size(8.dp)
                            .clip(CircleShape)
                            .background(CallAcceptGreen)
                    )
                    Spacer(modifier = Modifier.width(8.dp))
                    Text(
                        text = if (isGreeting) "Playing Greeting to Caller" else "Caller Recording Message",
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Medium,
                        color = TextSecondary
                    )
                }

                if (isSpeakerListening) {
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        modifier = Modifier
                            .clip(RoundedCornerShape(6.dp))
                            .background(Color(0x33007AFF))
                            .padding(horizontal = 6.dp, vertical = 2.dp)
                    ) {
                        Icon(
                            imageVector = Icons.Default.VolumeUp,
                            contentDescription = null,
                            tint = Color(0xFF007AFF),
                            modifier = Modifier.size(12.dp)
                        )
                        Spacer(modifier = Modifier.width(4.dp))
                        Text(
                            text = "Listening",
                            fontSize = 10.sp,
                            color = Color(0xFF007AFF),
                            fontWeight = FontWeight.Bold
                        )
                    }
                }
            }

            // Transcript Scroll View
            LazyColumn(
                state = listState,
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
                    .padding(vertical = 12.dp)
            ) {
                item {
                    if (transcript.isEmpty()) {
                        Box(
                            modifier = Modifier
                                .fillMaxWidth()
                                .height(200.dp),
                            contentAlignment = Alignment.Center
                        ) {
                            Text(
                                text = if (isGreeting) "Greeting caller through cellular uplinkâ€¦" else "Waiting for caller to speakâ€¦",
                                color = TextSecondary,
                                fontSize = 16.sp,
                                textAlign = TextAlign.Center
                            )
                        }
                    } else {
                        Text(
                            text = transcript,
                            color = TextPrimary,
                            fontSize = 22.sp,
                            lineHeight = 30.sp,
                            fontWeight = FontWeight.Medium
                        )
                    }
                }
                item {
                    Spacer(modifier = Modifier.height(16.dp))
                }
            }

            // Footer note
            Text(
                text = "Live Voicemail recording caller. Tap Pick Up to answer.",
                fontSize = 11.sp,
                color = TextSecondary,
                textAlign = TextAlign.Center,
                modifier = Modifier.fillMaxWidth()
            )
        }
    }
}

@Composable
private fun AddCallOverlay(
    onClose: () -> Unit,
    onPlaceCall: (String) -> Unit
) {
    var inputNumber by remember { mutableStateOf("") }
    val keys = listOf(
        listOf("1" to "", "2" to "ABC", "3" to "DEF"),
        listOf("4" to "GHI", "5" to "JKL", "6" to "MNO"),
        listOf("7" to "PQRS", "8" to "TUV", "9" to "WXYZ"),
        listOf("*" to "", "0" to "+", "#" to "")
    )

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color(0xE60A0A0C))
            .padding(24.dp)
    ) {
        Column(
            modifier = Modifier.fillMaxSize(),
            horizontalAlignment = Alignment.CenterHorizontally,
            verticalArrangement = Arrangement.SpaceBetween
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = "Add Call (Conference)",
                    fontSize = 18.sp,
                    fontWeight = FontWeight.Bold,
                    color = TextPrimary
                )
                Text(
                    text = "Cancel",
                    fontSize = 16.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = Color(0xFF007AFF),
                    modifier = Modifier
                        .clickable { onClose() }
                        .padding(8.dp)
                )
            }

            // Display dialed number
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 16.dp, vertical = 8.dp),
                horizontalArrangement = Arrangement.Center,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = inputNumber.ifEmpty { "Enter Number" },
                    fontSize = if (inputNumber.isEmpty()) 20.sp else 30.sp,
                    fontWeight = FontWeight.Bold,
                    color = if (inputNumber.isEmpty()) TextSecondary else TextPrimary,
                    textAlign = TextAlign.Center,
                    modifier = Modifier.weight(1f)
                )

                if (inputNumber.isNotEmpty()) {
                    Icon(
                        imageVector = Icons.Default.Backspace,
                        contentDescription = "Backspace",
                        tint = TextSecondary,
                        modifier = Modifier
                            .size(28.dp)
                            .clickable {
                                inputNumber = inputNumber.dropLast(1)
                            }
                    )
                }
            }

            // Dialpad grid
            Column(
                modifier = Modifier.fillMaxWidth(),
                verticalArrangement = Arrangement.spacedBy(14.dp),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                for (row in keys) {
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.SpaceEvenly
                    ) {
                        for ((digit, sub) in row) {
                            DtmfKeyButton(
                                digit = digit,
                                subtext = sub,
                                onPress = {
                                    inputNumber += digit
                                },
                                onRelease = {}
                            )
                        }
                    }
                }
            }

            // Green Call Button
            Box(
                modifier = Modifier
                    .size(68.dp)
                    .clip(CircleShape)
                    .background(if (inputNumber.isNotBlank()) CallAcceptGreen else Color.DarkGray)
                    .clickable(enabled = inputNumber.isNotBlank()) {
                        onPlaceCall(inputNumber)
                    },
                contentAlignment = Alignment.Center
            ) {
                Icon(
                    imageVector = Icons.Default.Call,
                    contentDescription = "Call",
                    tint = Color.White,
                    modifier = Modifier.size(32.dp)
                )
            }
        }
    }
}

/**
 * Direct-Boot aware persistent configuration helper for system readiness,
 * battery survival exemptions, and OEM setup states.
 *
 * Uses [Context.createDeviceProtectedStorageContext] so preferences can be accessed
 * immediately after device restart before the user unlocks the lockscreen.
 */
object SetupPreferences {
    private const val TAG = "SetupPreferences"
    private const val PREFS_NAME = "voicemail_setup_prefs"

    const val KEY_BATTERY_EXEMPT_CONFIRMED = "battery_exempt_confirmed"
    const val KEY_ACCESSIBILITY_CONFIRMED = "accessibility_confirmed"
    const val KEY_DIALER_CONFIRMED = "dialer_confirmed"
    const val KEY_RUNTIME_PERMS_CONFIRMED = "runtime_perms_confirmed"

    fun getPrefs(context: Context): SharedPreferences {
        val storageContext = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N) {
            context.createDeviceProtectedStorageContext()
        } else {
            context
        }
        return storageContext.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
    }

    fun isBatteryExemptConfirmed(context: Context): Boolean {
        return getPrefs(context).getBoolean(KEY_BATTERY_EXEMPT_CONFIRMED, false)
    }

    fun setBatteryExemptConfirmed(context: Context, confirmed: Boolean) {
        getPrefs(context).edit().putBoolean(KEY_BATTERY_EXEMPT_CONFIRMED, confirmed).apply()
    }

    fun isAccessibilityConfirmed(context: Context): Boolean {
        return getPrefs(context).getBoolean(KEY_ACCESSIBILITY_CONFIRMED, false)
    }

    fun setAccessibilityConfirmed(context: Context, confirmed: Boolean) {
        getPrefs(context).edit().putBoolean(KEY_ACCESSIBILITY_CONFIRMED, confirmed).apply()
    }

    fun isDialerConfirmed(context: Context): Boolean {
        return getPrefs(context).getBoolean(KEY_DIALER_CONFIRMED, false)
    }

    fun setDialerConfirmed(context: Context, confirmed: Boolean) {
        getPrefs(context).edit().putBoolean(KEY_DIALER_CONFIRMED, confirmed).apply()
    }

    fun isRuntimePermsConfirmed(context: Context): Boolean {
        return getPrefs(context).getBoolean(KEY_RUNTIME_PERMS_CONFIRMED, false)
    }

    fun setRuntimePermsConfirmed(context: Context, confirmed: Boolean) {
        getPrefs(context).edit().putBoolean(KEY_RUNTIME_PERMS_CONFIRMED, confirmed).apply()
    }

    /**
     * Opens OEM-specific autostart and background power management settings
     * (specifically targeting Vivo FuntouchOS / iQOO / ColorOS / MIUI).
     */
    fun openOemBackgroundSettings(context: Context): Boolean {
        val intents = listOf(
            // Vivo & iQOO High Background Power Consumption
            Intent().setComponent(ComponentName("com.iqoo.powersaving", "com.iqoo.powersaving.PowerSavingManagerActivity")),
            Intent().setComponent(ComponentName("com.iqoo.powersaving", "com.iqoo.powersaving.HighPowerManagerActivity")),
            // Vivo & iQOO Autostart & Background Startup
            Intent().setComponent(ComponentName("com.vivo.permissionmanager", "com.vivo.permissionmanager.activity.BgStartUpManagerActivity")),
            Intent().setComponent(ComponentName("com.vivo.permissionmanager", "com.vivo.permissionmanager.activity.PurviewTabActivity")),
            // Xiaomi / MIUI Autostart
            Intent().setComponent(ComponentName("com.miui.securitycenter", "com.miui.permcenter.autostart.AutoStartManagementActivity")),
            // Oppo / Realme Autostart
            Intent().setComponent(ComponentName("com.coloros.safecenter", "com.coloros.safecenter.permission.startup.StartupAppListActivity")),
            Intent().setComponent(ComponentName("com.oppo.safe", "com.oppo.safe.permission.startup.StartupAppListActivity")),
            // Standard Battery Optimization Settings
            Intent(Settings.ACTION_IGNORE_BATTERY_OPTIMIZATION_SETTINGS),
            // Fallback App Details
            Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS).apply {
                data = Uri.parse("package:${context.packageName}")
            }
        )

        for (intent in intents) {
            try {
                intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                context.startActivity(intent)
                Log.i(TAG, "Successfully launched OEM background settings: ${intent.component ?: intent.action}")
                return true
            } catch (t: Throwable) {
                // Try next candidate
            }
        }
        return false
    }
