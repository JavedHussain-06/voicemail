/**
 * Generates and manages the telephony greeting WAV audio files.
 *
 * Guarantees a valid 16,000 Hz, 16-bit Mono RIFF/WAVE greeting exists on disk at files/default_greeting.wav.
 * Uses Android TextToSpeech to synthesize spoken voice audio and resamples to 16kHz with an 800Hz cue beep.
 */
object GreetingGenerator {

    private const val TAG = "GreetingGenerator"
    private const val GREETING_DIR_NAME = "greetings"
    const val DEFAULT_GREETING_FILE_NAME = "default_greeting.wav"
    const val CUSTOM_GREETING_FILE_NAME = "custom_greeting.wav"
    const val GREETING_SPOKEN_TEXT = "Hello, I cannot take your call right now. Please leave your message after the beep."

    const val SAMPLE_RATE_HZ = 16000
    const val CHANNELS = 1 // Mono for cellular uplink
    const val BITS_PER_SAMPLE = 16

    @Volatile
    private var isGeneratingTts = false

    /**
     * Retrieves or automatically synthesizes the active greeting WAV file.
     * Prefers custom greeting if created; otherwise ensures default_greeting.wav exists.
     */
    @Synchronized
    fun getActiveGreetingFile(context: Context): File {
        val rootGreeting = File(context.filesDir, DEFAULT_GREETING_FILE_NAME)
        val dir = File(context.filesDir, GREETING_DIR_NAME).apply { mkdirs() }
        val customGreeting = File(dir, CUSTOM_GREETING_FILE_NAME)
        if (customGreeting.exists() && customGreeting.length() > 44) {
            return customGreeting
        }

        val dirGreeting = File(dir, DEFAULT_GREETING_FILE_NAME)

        // Check if root greeting exists and is valid
        if (rootGreeting.exists() && rootGreeting.length() >= 64000) {
            // Keep dirGreeting in sync
            if (!dirGreeting.exists() || dirGreeting.length() != rootGreeting.length()) {
                runCatching { rootGreeting.copyTo(dirGreeting, overwrite = true) }
            }
            stageToTmpIfPossible(rootGreeting)
            return rootGreeting
        }

        // Check if dirGreeting exists and is valid
        if (dirGreeting.exists() && dirGreeting.length() >= 64000) {
            runCatching { dirGreeting.copyTo(rootGreeting, overwrite = true) }
            stageToTmpIfPossible(rootGreeting)
            return rootGreeting
        }

        // Generate immediate fallback chime first so an active file is guaranteed to exist right now
        generateImmediateFallbackGreeting(rootGreeting)
        runCatching { rootGreeting.copyTo(dirGreeting, overwrite = true) }
        stageToTmpIfPossible(rootGreeting)

        // Trigger asynchronous high-fidelity TTS spoken voice generation
        generateVoiceGreetingTts(context)

        return rootGreeting
    }

    /**
     * Attempts to stage greeting directly into /data/local/tmp/voicemail_greeting.wav
     */
    fun stageToTmpIfPossible(sourceFile: File) {
        try {
            val tmpFile = File("/data/local/tmp/voicemail_greeting.wav")
            sourceFile.copyTo(tmpFile, overwrite = true)
            tmpFile.setReadable(true, false)
            Log.i(TAG, "Staged greeting directly to ${tmpFile.absolutePath} (${tmpFile.length()} bytes)")
        } catch (t: Throwable) {
            Log.d(TAG, "Direct copy to /data/local/tmp deferred to Shizuku/ADB: ${t.message}")
        }
    }

    /**
     * Creates an AssetFileDescriptor wrapping the active greeting WAV file.
     */
    fun getGreetingAssetFileDescriptor(context: Context): AssetFileDescriptor {
        val file = getActiveGreetingFile(context)
        val pfd = ParcelFileDescriptor.open(file, ParcelFileDescriptor.MODE_READ_ONLY)
        return AssetFileDescriptor(pfd, 0, file.length())
    }

    /**
     * Synthesizes spoken voice greeting using Android TextToSpeech, resamples to 16,000 Hz,
     * appends an 800Hz beep tone, and saves to files/default_greeting.wav.
     */
    fun generateVoiceGreetingTts(context: Context, onComplete: ((Boolean) -> Unit)? = null) {
        if (isGeneratingTts) {
            Log.d(TAG, "TTS voice generation already in progress, skipping duplicate call.")
            onComplete?.invoke(false)
            return
        }
        isGeneratingTts = true

        val appContext = context.applicationContext
        val mainHandler = Handler(Looper.getMainLooper())

        mainHandler.post {
            var tts: TextToSpeech? = null
            tts = TextToSpeech(appContext) { status ->
                if (status != TextToSpeech.SUCCESS) {
                    Log.e(TAG, "TextToSpeech initialization failed with status $status")
                    isGeneratingTts = false
                    tts?.shutdown()
                    onComplete?.invoke(false)
                    return@TextToSpeech
                }

                tts?.let { engine ->
                    try {
                        engine.language = Locale.US

                        val tempTtsFile = File(appContext.cacheDir, "temp_tts_greeting.wav")
                        if (tempTtsFile.exists()) tempTtsFile.delete()

                        val utteranceId = "voicemail_greeting_tts"
                        engine.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
                            override fun onStart(id: String?) {
                                Log.i(TAG, "TextToSpeech started synthesizing spoken voice greeting...")
                            }

                            override fun onDone(id: String?) {
                                Log.i(TAG, "TextToSpeech finished synthesis. Processing WAV audio (${tempTtsFile.length()} bytes)...")
                                try {
                                    val success = processTtsWavTo16kWithBeep(tempTtsFile, appContext)
                                    isGeneratingTts = false
                                    engine.shutdown()
                                    onComplete?.invoke(success)
                                } catch (t: Throwable) {
                                    Log.e(TAG, "Error processing TTS WAV output", t)
                                    isGeneratingTts = false
                                    engine.shutdown()
                                    onComplete?.invoke(false)
                                } finally {
                                    tempTtsFile.delete()
                                }
                            }

                            @Deprecated("Deprecated in Java")
                            override fun onError(id: String?) {
                                Log.e(TAG, "TextToSpeech synthesis error for utterance: $id")
                                isGeneratingTts = false
                                engine.shutdown()
                                onComplete?.invoke(false)
                            }

                            override fun onError(id: String?, errorCode: Int) {
                                Log.e(TAG, "TextToSpeech synthesis error $errorCode for utterance: $id")
                                isGeneratingTts = false
                                engine.shutdown()
                                onComplete?.invoke(false)
                            }
                        })

                        val params = Bundle().apply {
                            putFloat(TextToSpeech.Engine.KEY_PARAM_VOLUME, 1.0f)
                            putFloat(TextToSpeech.Engine.KEY_PARAM_PAN, 0.0f)
                        }

                        val result = engine.synthesizeToFile(
                            GREETING_SPOKEN_TEXT,
                            params,
                            tempTtsFile,
                            utteranceId
                        )
                        Log.i(TAG, "TextToSpeech.synthesizeToFile returned $result for '$GREETING_SPOKEN_TEXT'")
                        if (result != TextToSpeech.SUCCESS) {
                            isGeneratingTts = false
                            engine.shutdown()
                            onComplete?.invoke(false)
                        }
                    } catch (t: Throwable) {
                        Log.e(TAG, "Exception during TTS setup", t)
                        isGeneratingTts = false
                        engine.shutdown()
                        onComplete?.invoke(false)
                    }
                }
            }
        }
    }

    /**
     * Reads the raw TTS output WAV, parses its source sample rate (e.g. 22050 or 24000 Hz),
     * resamples to 16,000 Hz Mono 16-bit PCM, appends a 500ms 800Hz beep (amplitude = 20000),
     * and saves to files/default_greeting.wav.
     */
    private fun processTtsWavTo16kWithBeep(sourceWav: File, context: Context): Boolean {
        if (!sourceWav.exists() || sourceWav.length() < 44) {
            Log.e(TAG, "TTS source WAV missing or invalid: ${sourceWav.length()} bytes")
            return false
        }

        val fileBytes = sourceWav.readBytes()
        if (fileBytes.size < 44) return false

        // Parse WAV Header
        val headerBuf = ByteBuffer.wrap(fileBytes).order(ByteOrder.LITTLE_ENDIAN)
        val riff = String(fileBytes, 0, 4)
        val wave = String(fileBytes, 8, 4)
        if (riff != "RIFF" || wave != "WAVE") {
            Log.w(TAG, "TTS output is not standard RIFF/WAVE ($riff/$wave)")
        }

        // Find "data" chunk
        var dataOffset = 44
        var dataLen = fileBytes.size - 44
        var sourceSampleRate = 22050
        var sourceChannels = 1
        var bitsPerSample = 16

        var offset = 12
        while (offset < fileBytes.size - 8) {
            val chunkId = String(fileBytes, offset, 4)
            val chunkSize = ByteBuffer.wrap(fileBytes, offset + 4, 4).order(ByteOrder.LITTLE_ENDIAN).int
            if (chunkId == "fmt ") {
                val format = ByteBuffer.wrap(fileBytes, offset + 8, 2).order(ByteOrder.LITTLE_ENDIAN).short
                sourceChannels = ByteBuffer.wrap(fileBytes, offset + 10, 2).order(ByteOrder.LITTLE_ENDIAN).short.toInt()
                sourceSampleRate = ByteBuffer.wrap(fileBytes, offset + 12, 4).order(ByteOrder.LITTLE_ENDIAN).int
                bitsPerSample = ByteBuffer.wrap(fileBytes, offset + 22, 2).order(ByteOrder.LITTLE_ENDIAN).short.toInt()
                Log.i(TAG, "Parsed TTS WAV fmt: format=$format, channels=$sourceChannels, sampleRate=$sourceSampleRate, bits=$bitsPerSample")
            } else if (chunkId == "data") {
                dataOffset = offset + 8
                dataLen = chunkSize
                break
            }
            offset += 8 + chunkSize
        }

        if (sourceSampleRate <= 0) sourceSampleRate = 22050
        if (sourceChannels <= 0) sourceChannels = 1
        if (bitsPerSample <= 0) bitsPerSample = 16

        // Extract PCM 16-bit mono samples from source
        val bytesPerSample = bitsPerSample / 8
        val frameSize = bytesPerSample * sourceChannels
        val totalFrames = dataLen / frameSize

        val sourcePcm = ShortArray(totalFrames)
        var byteIdx = dataOffset
        for (i in 0 until totalFrames) {
            if (byteIdx + frameSize > fileBytes.size) break
            var monoSample = 0
            for (ch in 0 until sourceChannels) {
                val s = if (bitsPerSample == 16) {
                    (fileBytes[byteIdx + ch * 2].toInt() and 0xFF or
                            (fileBytes[byteIdx + ch * 2 + 1].toInt() shl 8)).toShort()
                } else {
                    ((fileBytes[byteIdx + ch].toInt() and 0xFF - 128) shl 8).toShort()
                }
                monoSample += s
            }
            sourcePcm[i] = (monoSample / sourceChannels).toShort()
            byteIdx += frameSize
        }

        Log.i(TAG, "Extracted ${sourcePcm.size} source samples at $sourceSampleRate Hz. Resampling to $SAMPLE_RATE_HZ Hz...")

        // Resample from sourceSampleRate to 16,000 Hz using linear interpolation
        val resampledCount = ((sourcePcm.size.toLong() * SAMPLE_RATE_HZ) / sourceSampleRate).toInt()
        val resampledPcm = ShortArray(resampledCount)
        val ratio = sourceSampleRate.toDouble() / SAMPLE_RATE_HZ.toDouble()

        for (i in 0 until resampledCount) {
            val srcPos = i * ratio
            val srcIdx = srcPos.toInt()
            val frac = srcPos - srcIdx
            val s0 = if (srcIdx < sourcePcm.size) sourcePcm[srcIdx] else 0
            val s1 = if (srcIdx + 1 < sourcePcm.size) sourcePcm[srcIdx + 1] else s0
            val interpolated = (s0 * (1.0 - frac) + s1 * frac).toInt().coerceIn(-32768, 32767)
            resampledPcm[i] = interpolated.toShort()
        }

        // Generate 500ms 800Hz sine-wave beep tone (amplitude = 20000)
        val beepDurationMs = 500
        val beepSampleCount = (SAMPLE_RATE_HZ * beepDurationMs) / 1000
        val beepSamples = ShortArray(beepSampleCount)
        val beepAmp = 20000.0

        for (i in 0 until beepSampleCount) {
            val attack = if (i < 160) (i / 160.0) else 1.0
            val release = if (i > beepSampleCount - 160) ((beepSampleCount - i) / 160.0) else 1.0
            val angle = 2.0 * PI * i * 800.0 / SAMPLE_RATE_HZ
            val s = (sin(angle) * beepAmp * attack * release).toInt().coerceIn(-32768, 32767)
            beepSamples[i] = s.toShort()
        }

        // 100ms silence between voice and beep, and 150ms tail silence
        val pauseSamples = ShortArray((SAMPLE_RATE_HZ * 0.100).toInt())
        val tailSamples = ShortArray((SAMPLE_RATE_HZ * 0.150).toInt())

        val totalFinalSamples = resampledPcm.size + pauseSamples.size + beepSamples.size + tailSamples.size
        val finalPcmBytes = ByteArray(totalFinalSamples * 2)
        val buf = ByteBuffer.wrap(finalPcmBytes).order(ByteOrder.LITTLE_ENDIAN)

        for (s in resampledPcm) buf.putShort(s)
        for (s in pauseSamples) buf.putShort(0)
        for (s in beepSamples) buf.putShort(s)
        for (s in tailSamples) buf.putShort(0)

        // Write final 16kHz mono WAV file
        val targetFile = File(context.filesDir, DEFAULT_GREETING_FILE_NAME)
        val dirGreeting = File(File(context.filesDir, GREETING_DIR_NAME), DEFAULT_GREETING_FILE_NAME)

        FileOutputStream(targetFile).use { fos ->
            writeWavHeader(
                out = fos,
                totalAudioLen = finalPcmBytes.size.toLong(),
                sampleRate = SAMPLE_RATE_HZ,
                channels = CHANNELS,
                bitsPerSample = BITS_PER_SAMPLE
            )
            fos.write(finalPcmBytes)
            fos.flush()
        }

        runCatching { targetFile.copyTo(dirGreeting, overwrite = true) }
        stageToTmpIfPossible(targetFile)

        // Read back PCM samples and verify peak amplitude
        var maxSample = 0
        for (i in 0 until totalFinalSamples) {
            val s = kotlin.math.abs(buf.getShort(i * 2).toInt())
            if (s > maxSample) maxSample = s
        }

        val durationSec = totalFinalSamples.toFloat() / SAMPLE_RATE_HZ
        Log.d(TAG, "WAV ready: size=${targetFile.length()} peak=$maxSample")
        Log.i(TAG, "Spoken voice greeting generated with 800Hz beep! File: ${targetFile.absolutePath}, Size: ${targetFile.length()} bytes, Duration: ${durationSec}s, Peak: $maxSample")
        return targetFile.length() > 64000 && maxSample > 5000
    }

    /**
     * Programmatically synthesizes an immediate 4-second 16kHz Mono WAV greeting
     * (warm telecom chime + speech cadence pulses + 800Hz beep) so the system is NEVER empty.
     */
    fun generateImmediateFallbackGreeting(outputFile: File) {
        Log.i(TAG, "Synthesizing immediate fallback 16kHz mono WAV greeting at: ${outputFile.absolutePath}")
        val pcmData = ArrayList<Short>()

        // 1. Initial comfort silence
        appendSilence(pcmData, 150)

        // 2. Warm opening chime (C5 -> E5 -> G5 -> C6)
        appendChimeNote(pcmData, frequencyHz = 523.25, durationMs = 180, peakAmplitude = 0.55)
        appendChimeNote(pcmData, frequencyHz = 659.25, durationMs = 180, peakAmplitude = 0.60)
        appendChimeNote(pcmData, frequencyHz = 783.99, durationMs = 220, peakAmplitude = 0.65)
        appendChimeNote(pcmData, frequencyHz = 1046.50, durationMs = 380, peakAmplitude = 0.70)

        appendSilence(pcmData, 250)

        // 3. Cadence pulses
        appendSpeechPulse(pcmData, durationMs = 220, baseFreq = 300.0) // "Please"
        appendSilence(pcmData, 80)
        appendSpeechPulse(pcmData, durationMs = 200, baseFreq = 340.0) // "leave"
        appendSilence(pcmData, 60)
        appendSpeechPulse(pcmData, durationMs = 160, baseFreq = 320.0) // "your"
        appendSilence(pcmData, 60)
        appendSpeechPulse(pcmData, durationMs = 300, baseFreq = 280.0) // "message"
        appendSilence(pcmData, 120)
        appendSpeechPulse(pcmData, durationMs = 180, baseFreq = 320.0) // "after"
        appendSilence(pcmData, 60)
        appendSpeechPulse(pcmData, durationMs = 160, baseFreq = 350.0) // "the"
        appendSilence(pcmData, 60)
        appendSpeechPulse(pcmData, durationMs = 350, baseFreq = 400.0) // "tone."

        appendSilence(pcmData, 300)

        // 4. Standard 800 Hz beep tone (400ms, amplitude 16000)
        appendSineTone(pcmData, frequencyHz = 800.0, durationMs = 400, amplitude = 0.60)

        appendSilence(pcmData, 150)

        val pcmBytes = ByteArray(pcmData.size * 2)
        val byteBuffer = ByteBuffer.wrap(pcmBytes).order(ByteOrder.LITTLE_ENDIAN)
        for (sample in pcmData) {
            byteBuffer.putShort(sample)
        }

        outputFile.parentFile?.mkdirs()
        FileOutputStream(outputFile).use { fos ->
            writeWavHeader(
                out = fos,
                totalAudioLen = pcmBytes.size.toLong(),
                sampleRate = SAMPLE_RATE_HZ,
                channels = CHANNELS,
                bitsPerSample = BITS_PER_SAMPLE
            )
            fos.write(pcmBytes)
            fos.flush()
        }
        Log.i(TAG, "Immediate fallback greeting synthesized: ${outputFile.length()} bytes")
    }

    private fun appendSilence(buffer: ArrayList<Short>, durationMs: Int) {
        val count = (SAMPLE_RATE_HZ * (durationMs / 1000.0)).toInt()
        for (i in 0 until count) {
            buffer.add(0)
        }
    }

    private fun appendSineTone(buffer: ArrayList<Short>, frequencyHz: Double, durationMs: Int, amplitude: Double) {
        val totalSamples = (SAMPLE_RATE_HZ * (durationMs / 1000.0)).toInt()
        val maxAmp = 32767.0 * amplitude
        val attackSamples = (SAMPLE_RATE_HZ * 0.01).toInt()
        val releaseSamples = (SAMPLE_RATE_HZ * 0.02).toInt()

        for (i in 0 until totalSamples) {
            var env = 1.0
            if (i < attackSamples) {
                env = i.toDouble() / attackSamples
            } else if (i > totalSamples - releaseSamples) {
                env = (totalSamples - i).toDouble() / releaseSamples
            }
            val angle = 2.0 * PI * i * frequencyHz / SAMPLE_RATE_HZ
            val sampleVal = (sin(angle) * maxAmp * env).toInt().coerceIn(-32768, 32767)
            buffer.add(sampleVal.toShort())
        }
    }

    private fun appendChimeNote(buffer: ArrayList<Short>, frequencyHz: Double, durationMs: Int, peakAmplitude: Double) {
        val totalSamples = (SAMPLE_RATE_HZ * (durationMs / 1000.0)).toInt()
        val maxAmp = 32767.0 * peakAmplitude

        for (i in 0 until totalSamples) {
            val t = i.toDouble() / SAMPLE_RATE_HZ
            val progress = i.toDouble() / totalSamples
            val decay = kotlin.math.exp(-3.5 * progress)
            val attack = if (i < (SAMPLE_RATE_HZ * 0.015)) (i / (SAMPLE_RATE_HZ * 0.015)) else 1.0

            val fundamental = sin(2.0 * PI * frequencyHz * t)
            val overtone2 = 0.25 * sin(2.0 * PI * (frequencyHz * 2.0) * t)
            val overtone3 = 0.10 * sin(2.0 * PI * (frequencyHz * 3.0) * t)

            val sampleVal = ((fundamental + overtone2 + overtone3) * maxAmp * decay * attack)
                .toInt()
                .coerceIn(-32768, 32767)

            buffer.add(sampleVal.toShort())
        }
    }

    private fun appendSpeechPulse(buffer: ArrayList<Short>, durationMs: Int, baseFreq: Double) {
        val totalSamples = (SAMPLE_RATE_HZ * (durationMs / 1000.0)).toInt()
        val maxAmp = 32767.0 * 0.40

        for (i in 0 until totalSamples) {
            val t = i.toDouble() / SAMPLE_RATE_HZ
            val progress = i.toDouble() / totalSamples
            val env = sin(PI * progress)

            val f0 = sin(2.0 * PI * baseFreq * t)
            val f1 = 0.4 * sin(2.0 * PI * (baseFreq * 2.8) * t)
            val f2 = 0.2 * sin(2.0 * PI * (baseFreq * 5.2) * t)

            val sampleVal = ((f0 + f1 + f2) * maxAmp * env).toInt().coerceIn(-32768, 32767)
            buffer.add(sampleVal.toShort())
        }
    }

    /**
     * Writes standard 44-byte RIFF/WAVE header to an OutputStream.
     */
    fun writeWavHeader(
        out: OutputStream,
        totalAudioLen: Long,
        sampleRate: Int = SAMPLE_RATE_HZ,
        channels: Int = CHANNELS,
        bitsPerSample: Int = BITS_PER_SAMPLE
    ) {
        val totalDataLen = totalAudioLen + 36
        val byteRate = (sampleRate * channels * bitsPerSample / 8).toLong()
        val blockAlign = (channels * bitsPerSample / 8).toShort()

        val header = ByteArray(44)
        val buffer = ByteBuffer.wrap(header).order(ByteOrder.LITTLE_ENDIAN)

        buffer.put('R'.code.toByte())
        buffer.put('I'.code.toByte())
        buffer.put('F'.code.toByte())
        buffer.put('F'.code.toByte())
        buffer.putInt(totalDataLen.toInt())
        buffer.put('W'.code.toByte())
        buffer.put('A'.code.toByte())
        buffer.put('V'.code.toByte())
        buffer.put('E'.code.toByte())

        buffer.put('f'.code.toByte())
        buffer.put('m'.code.toByte())
        buffer.put('t'.code.toByte())
        buffer.put(' '.code.toByte())
        buffer.putInt(16) // PCM subchunk1 size
        buffer.putShort(1) // PCM format
        buffer.putShort(channels.toShort())
        buffer.putInt(sampleRate)
        buffer.putInt(byteRate.toInt())
        buffer.putShort(blockAlign)
        buffer.putShort(bitsPerSample.toShort())

        buffer.put('d'.code.toByte())
        buffer.put('a'.code.toByte())
        buffer.put('t'.code.toByte())
        buffer.put('a'.code.toByte())
        buffer.putInt(totalAudioLen.toInt())

        out.write(header, 0, 44)
    }

    /**
     * Updates an existing WAV file's 44-byte header in place.
     */
    fun finalizeWavHeaderInPlace(file: File) {
        if (!file.exists()) return
        val totalFileLen = file.length()
        if (totalFileLen < 44) return

        val audioPayloadLen = totalFileLen - 44
        val totalDataLen = audioPayloadLen + 36

        RandomAccessFile(file, "rw").use { raf ->
            raf.seek(4)
            raf.write(intToLittleEndian(totalDataLen.toInt()))
            raf.seek(40)
            raf.write(intToLittleEndian(audioPayloadLen.toInt()))
        }
        Log.i(TAG, "Finalized WAV header for ${file.name}: payload=${audioPayloadLen}B, total=${totalFileLen}B")
    }

    private fun intToLittleEndian(value: Int): ByteArray {
        return byteArrayOf(
            (value and 0xFF).toByte(),
            ((value shr 8) and 0xFF).toByte(),

                Log.d(TAG, "Starting Sherpa-ONNX Zipformer model extraction and initialization...")
                val targetDir = File(context.filesDir, MODEL_DIR).apply { mkdirs() }
                val requiredFiles = listOf(
                    "encoder.int8.onnx",
                    "decoder.onnx",
                    "joiner.onnx",

                            context.assets.open("$MODEL_DIR/$name").use { input ->
                                FileOutputStream(dest).use { output ->
                                    input.copyTo(output)
                                }

                            Log.d(TAG, "Extracted $name (${dest.length()} bytes)")
                        } catch (e: Exception) {
                            Log.w(TAG, "Failed extracting $name from assets: ${e.message}")
                        }
                    } else {
                        Log.d(TAG, "Model file $name already exists (${dest.length()} bytes)")
                    }
                    if (dest.exists() && dest.length() > 0L) {
                        modelFiles[name] = dest
                    }
                }

                val decoderFile = modelFiles["decoder.onnx"]
                val joinerFile = modelFiles["joiner.onnx"]
                val tokensFile = modelFiles["tokens.txt"]

                    Log.d(TAG, "Configuring Zipformer transducer...")

                        encoder = encoderFile.absolutePath,
                        decoder = decoderFile.absolutePath,
                        joiner = joinerFile.absolutePath

                        transducer = transducerConfig,
                        tokens = tokensFile.absolutePath,
                        numThreads = 2,

                        debug = false

    /**
     * Consumer loop: receives 16kHz PCM chunks, writes raw bytes, feeds Sherpa-ONNX stream.
     */
    private suspend fun consumeAudioChunks(channel: Channel<ShortArray>) {
        val stream = activeStream
        val rec = recognizer
        val byteBuffer = ByteBuffer.allocate(1600 * 2).order(ByteOrder.LITTLE_ENDIAN)

        for (chunk in channel) {
            // 1. Write raw PCM to temporary disk file
            byteBuffer.clear()
            for (sample in chunk) {
                byteBuffer.putShort(sample)
            }
            try {
                tempPcmOutputStream?.write(byteBuffer.array(), 0, byteBuffer.position())
            } catch (e: Exception) {
                Log.e(TAG, "Error writing raw PCM chunk to disk", e)
            }

            // 2. Calculate RMS Audio Level for real-time visual waveform animation
            val rms = calculateRms(chunk)
            _audioRmsLevel.value = rms

            // 3. Normalize samples to FloatArray [-1.0f, 1.0f]
            val floatSamples = FloatArray(chunk.size)
            for (i in chunk.indices) {
                floatSamples[i] = chunk[i] / 32768.0f
            }

            // 4. Feed Sherpa-ONNX OnlineStream
            if (rec != null && stream != null) {
                stream.acceptWaveform(floatSamples, SAMPLE_RATE_HZ)

                while (rec.isReady(stream)) {
                    rec.decode(stream)
                }

                val currentResult = rec.getResult(stream)
                val partialText = currentResult.text.trim()

                if (rec.isEndpoint(stream)) {
                    if (partialText.isNotEmpty()) {
                        if (accumulatedSentences.isNotEmpty()) {
                            accumulatedSentences.append(" ")
                        }
                        accumulatedSentences.append(partialText)
                    }
                    rec.reset(stream)
                    _liveTranscript.value = accumulatedSentences.toString()
                } else {
                    val fullDisplay = if (accumulatedSentences.isEmpty()) {
                        partialText
                    } else if (partialText.isEmpty()) {
                        accumulatedSentences.toString()
                    } else {
                        "$accumulatedSentences $partialText"
                    }
                    _liveTranscript.value = fullDisplay
                }
            } else {
                // Fallback simulation when model weights are not loaded: speech activity indicators
                if (rms > 0.15f) {
                    val current = _liveTranscript.value
                    if (!current.contains("â€¦")) {
                        _liveTranscript.value = if (current.isEmpty()) "Speakingâ€¦" else "$current â€¦"
                    }
                }
            }
        }

        Log.i(TAG, "Consumer coroutine completed chunk processing loop.")
    }

     */
    fun submitPcmChunk(pcm16kHzSamples: ShortArray) {
        if (!isStopping.get()) {
            pcmChannel?.trySend(pcm16kHzSamples)
        }
    }

    /**
     * Finalizes the voicemail session:
     * - Shuts down audio channel and awaits consumer coroutine completion
     * - Wraps PCM file with 44-byte RIFF/WAVE header
     * - Stores VoicemailEntity in Room database (if saveToRepository == true)
     */
    suspend fun stopSession(isPickedUp: Boolean, saveToRepository: Boolean = true): VoicemailEntity? = withContext(Dispatchers.IO) {
        if (!_isScreeningActive.value) {
            return@withContext null
        }
        isStopping.set(true)
        _isScreeningActive.value = false

        // Close channel and wait for background consumer to finish writing
        pcmChannel?.close()
        consumerJob?.join()
        pcmChannel = null
        consumerJob = null

        // Release Sherpa stream
        try {
            activeStream?.let { s ->
                recognizer?.let { r ->
                    // Flush any remaining samples
                    while (r.isReady(s)) {
                        r.decode(s)
                    }
                    val finalResult = r.getResult(s)
                    if (finalResult.text.isNotBlank()) {
                        if (accumulatedSentences.isNotEmpty()) {
                            accumulatedSentences.append(" ")
                        }
                        accumulatedSentences.append(finalResult.text.trim())
                    }
                }
                s.release()
            }
        } catch (t: Throwable) {
            Log.w(TAG, "Error releasing Sherpa-ONNX stream", t)
        } finally {
            activeStream = null
        }

        // Close PCM output stream
        try {
            tempPcmOutputStream?.flush()
            tempPcmOutputStream?.close()
        } catch (ignored: Throwable) {
        } finally {
            tempPcmOutputStream = null
        }

        val pcm = tempPcmFile
        tempPcmFile = null

        if (!saveToRepository) {
            pcm?.delete()
            Log.i(TAG, "stopSession executed with saveToRepository=false (Self-Test mode). Temp files purged.")
            return@withContext null
        }

                    }

                }

            } catch (e: Exception) {

            }
        }

            isRead = false,
            isPickedUp = isPickedUp

        val id = repository.addVoicemail(entity)
        val persisted = entity.copy(id = id)

        _liveTranscript.value = finalTranscript
        _audioRmsLevel.value = 0f

        Log.i(TAG, "Voicemail recording saved successfully (ID: $id, Duration: ${durationSec}s, Caller: $currentPhoneNumber)")
        persisted
    }

    private fun calculateRms(samples: ShortArray): Float {
        if (samples.isEmpty()) return 0f
        var sumSquares = 0.0
        for (sample in samples) {
            val normalized = sample / 32768.0
            sumSquares += normalized * normalized
        }
        val meanSquare = sumSquares / samples.size
        return sqrt(meanSquare).toFloat().coerceIn(0f, 1f)
    }

class VoicemailInCallService : InCallService() {

    companion object {
        private const val TAG = "VoicemailInCallService"
        private const val INCOMING_NOTIFICATION_ID = 2001
        private const val AUTO_ANSWER_DELAY_MS = 8000L // 8-second unanswered countdown

        const val ACTION_TRIGGER_VOICEMAIL = "com.personal.voicemail.ACTION_TRIGGER_VOICEMAIL"
        const val ACTION_ANSWER_CALL = "com.personal.voicemail.ACTION_ANSWER_CALL"
        const val ACTION_PICK_UP = "com.personal.voicemail.ACTION_PICK_UP"
        const val ACTION_HANG_UP = "com.personal.voicemail.ACTION_HANG_UP"
    }

    private val serviceScope = CoroutineScope(Dispatchers.Main)
    private val appScope = CoroutineScope(SupervisorJob() + Dispatchers.IO)
    private val mainHandler = Handler(Looper.getMainLooper())

    private lateinit var audioManager: AudioManager
    private lateinit var shizukuAudioController: ShizukuAudioController
    private lateinit var transcriber: LiveVoicemailTranscriber

    private var activeCall: Call? = null
    private val allCalls = mutableListOf<Call>()
    private var proximityWakeLock: PowerManager.WakeLock? = null
    private var autoAnswerRunnable: Runnable? = null

    private val isScreeningSessionRunning = AtomicBoolean(false)
    private val hasSavedCurrentSession = AtomicBoolean(false)
    private val chunksReceivedCount = AtomicInteger(0)

    @Volatile
    private var isVoicemailScreeningActive = false
    private var inProcessGreetingPlayer: MediaPlayer? = null
    private var greetingTimeoutJob: Job? = null
    private val isGreetingFinished = AtomicBoolean(false)

    private var inProcessAudioRecord: AudioRecord? = null
    private var inProcessRecordJob: Job? = null

    private val callCallback = object : Call.Callback() {
        override fun onStateChanged(call: Call?, state: Int) {
            super.onStateChanged(call, state)
            Log.i(TAG, "Call onStateChanged: state=$state, screeningRunning=${isScreeningSessionRunning.get()}, screeningActive=$isVoicemailScreeningActive")
            when (state) {
                Call.STATE_RINGING -> {
                    TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.INCOMING_RINGING)
                    updateInCallNotification()
                }
                Call.STATE_DIALING, Call.STATE_CONNECTING -> {
                    TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.OUTGOING_DIALING)
                    updateInCallNotification()
                }
                Call.STATE_HOLDING -> {
                    TelecomCallManager.setIsOnHold(true)
                }
                Call.STATE_ACTIVE -> {
                    TelecomCallManager.setIsOnHold(false)
                    if (!isScreeningSessionRunning.get() && !isVoicemailScreeningActive) {
                        TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.ACTIVE_CALL)
                        acquireProximityWakeLock()
                        updateInCallNotification()
                    }
                }
                Call.STATE_DISCONNECTED, Call.STATE_DISCONNECTING -> {
                    if (call != null) {
                        allCalls.remove(call)
                        TelecomCallManager.updateCallsList(allCalls)
                    }
                    if (activeCall == call || allCalls.isEmpty()) {
                        handleCallDisconnected()
                    }
                }
            }
        }
    }

    /**
     * Receives 16kHz caller downlink PCM chunks from the privileged Shizuku VoicemailUserService.
     */
    private val audioChunkCallback = object : IAudioChunkCallback.Stub() {
        override fun onPcmChunkReceived(pcm16kHzBytes: ByteArray?) {
            if (pcm16kHzBytes != null && isVoicemailScreeningActive && isGreetingFinished.get()) {
                val count = chunksReceivedCount.incrementAndGet()
                val shortCount = pcm16kHzBytes.size / 2
                val shortArray = ShortArray(shortCount)
                ByteBuffer.wrap(pcm16kHzBytes).order(ByteOrder.LITTLE_ENDIAN).asShortBuffer().get(shortArray)

                // Boost quiet cellular downlink signal moderately without clipping
                var peak = 0
                for (i in shortArray.indices) {
                    val boosted = (shortArray[i].toInt() * 2).coerceIn(-32768, 32767)
                    val absVal = kotlin.math.abs(boosted)
                    if (absVal > peak) peak = absVal
                    shortArray[i] = boosted.toShort()
                }
                if (count <= 15 || count % 20 == 0) {
                    Log.d(TAG, "Privileged downlink chunk #$count received (${pcm16kHzBytes.size}B, boostedPeak=$peak)")
                }
                transcriber.submitPcmChunk(shortArray)
            }
        }

        override fun onGreetingFinished() {
            Log.i(TAG, "Privileged VoicemailUserService reported onGreetingFinished.")
            mainHandler.post {
                onGreetingPlaybackFinished()
            }
        }

        override fun onError(message: String?) {
            Log.e(TAG, "Privileged VoicemailUserService reported error: $message. Falling back to in-process downlink capture.")
            mainHandler.post {
                onGreetingPlaybackFinished()
                startInProcessDownlinkFallback()
            }
        }
    }

    override fun onCreate() {
        super.onCreate()
        audioManager = getSystemService(Context.AUDIO_SERVICE) as AudioManager
        shizukuAudioController = VoicemailApplication.shizukuController
        transcriber = VoicemailApplication.transcriber
        TelecomCallManager.currentInCallService = this
        // Pre-bind Shizuku UserService so it is hot when a call arrives
        shizukuAudioController.bindUserService()
        Log.i(TAG, "VoicemailInCallService created with shared singletons.")
    }

    override fun onDestroy() {
        super.onDestroy()
        cancelAutoAnswerTimer()
        releaseProximityWakeLock()
        stopInProcessGreetingFallback()
        stopInProcessAudioRecord()
        if (!hasSavedCurrentSession.get()) {
            stopAndSaveVoicemailSession(isPickedUp = false)
        }
        TelecomCallManager.currentInCallService = null
        allCalls.clear()
        Log.i(TAG, "VoicemailInCallService destroyed.")
    }

    override fun onCallAdded(call: Call) {
        call.registerCallback(callCallback)
        super.onCallAdded(call)

        activeCall = call
        if (!allCalls.contains(call)) {
            allCalls.add(call)
        }
        TelecomCallManager.updateCallsList(allCalls)

        hasSavedCurrentSession.set(false)
        isScreeningSessionRunning.set(false)
        isVoicemailScreeningActive = false

        val rawNumber = call.details?.handle?.schemeSpecificPart?.let { Uri.decode(it) }.orEmpty()
        val displayNumber = if (rawNumber.isNotBlank()) rawNumber else "Unknown Number"
        val callerName = call.details?.callerDisplayName?.takeIf { it.isNotBlank() }
            ?: resolveContactName(displayNumber)

        val (simLabel, simSlot) = resolveSimDetails(call.details?.accountHandle)

        val isOutgoing = call.state == Call.STATE_DIALING || call.state == Call.STATE_CONNECTING ||
                (call.details?.callDirection == Call.Details.DIRECTION_OUTGOING)

        Log.i(TAG, "onCallAdded: number=$displayNumber, caller=$callerName, SIM=$simLabel (slot=$simSlot), isOutgoing=$isOutgoing")

        val initialState = when {
            call.state == Call.STATE_RINGING -> TelecomCallManager.CallScreeningState.INCOMING_RINGING
            isOutgoing -> TelecomCallManager.CallScreeningState.OUTGOING_DIALING
            else -> TelecomCallManager.CallScreeningState.ACTIVE_CALL
        }

        TelecomCallManager.updateCall(
            call = call,
            name = callerName,
            number = displayNumber,
            simName = simLabel,
            simSlot = simSlot,
            state = initialState
        )

        updateInCallNotification()

        try {
            val activityIntent = Intent(this, InCallActivity::class.java).apply {
                addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_SINGLE_TOP or Intent.FLAG_ACTIVITY_CLEAR_TOP)
            }
            startActivity(activityIntent)
        } catch (t: Throwable) {
            Log.e(TAG, "Failed launching InCallActivity from onCallAdded", t)
        }

        if (call.state == Call.STATE_RINGING && !isOutgoing) {
            startAutoAnswerTimer()
        }
    }

    override fun onCallRemoved(call: Call) {
        super.onCallRemoved(call)
        call.unregisterCallback(callCallback)
        allCalls.remove(call)
        TelecomCallManager.updateCallsList(allCalls)
        if (activeCall == call || allCalls.isEmpty()) {
            handleCallDisconnected()
        }
    }

    override fun onCallAudioStateChanged(audioState: CallAudioState?) {
        super.onCallAudioStateChanged(audioState)
        // If voicemail screening is active, enforce hardware mute so the caller NEVER hears the user
        if (isVoicemailScreeningActive && audioState != null && !audioState.isMuted) {
            setMuted(true)
            audioManager.isMicrophoneMute = true
        }
        TelecomCallManager.updateAudioState(audioState)
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_TRIGGER_VOICEMAIL -> triggerLiveVoicemailScreening(activeCall)
            ACTION_ANSWER_CALL -> TelecomCallManager.answerCallNormally()
            ACTION_PICK_UP -> takeOverVoicemailCall()
            ACTION_HANG_UP -> activeCall?.disconnect()
        }
        return super.onStartCommand(intent, flags, startId)
    }

    fun placeSecondCall(number: String) {
        try {
            val uri = Uri.fromParts("tel", number, null)
            val tm = getSystemService(Context.TELECOM_SERVICE) as? TelecomManager
            tm?.placeCall(uri, null)
        } catch (t: Throwable) {
            Log.e(TAG, "Error placing second call: ${t.message}")
        }
    }

    private fun startAutoAnswerTimer() {
        cancelAutoAnswerTimer()
        val runnable = Runnable {
            val call = activeCall
            if (call != null && call.state == Call.STATE_RINGING) {
                Log.i(TAG, "Auto-answer 8s timer fired. Initiating silent Live Voicemail screening.")
                triggerLiveVoicemailScreening(call)
            }
        }
        autoAnswerRunnable = runnable
        mainHandler.postDelayed(runnable, AUTO_ANSWER_DELAY_MS)
    }

    private fun cancelAutoAnswerTimer() {
        autoAnswerRunnable?.let { mainHandler.removeCallbacks(it) }
        autoAnswerRunnable = null
    }

    /**
     * Answers the call silently, mutes the user's microphone and room speaker immediately,
     * and launches the privileged Shizuku voicemail greeting + downlink capture pipeline.
     */
    fun triggerLiveVoicemailScreening(call: Call? = activeCall) {
        val targetCall = call ?: return
        cancelAutoAnswerTimer()

        if (isScreeningSessionRunning.getAndSet(true)) {
            Log.d(TAG, "Screening session already running for current call.")
            return
        }

        serviceScope.launch(Dispatchers.Main) {
            Log.i(TAG, "Triggering Live Voicemail: Muting room audio & mic BEFORE answering.")
            isVoicemailScreeningActive = true
            TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.SCREENING_ANSWERING)
            TelecomCallManager.setScreeningSpeakerStateOnly(false)
            updateInCallNotification()

            // Pre-mute call stream & microphone before answering so there is zero audio pop/bleed
            enforceSilentScreeningAudioState()

            if (targetCall.state == Call.STATE_RINGING || targetCall.state == Call.STATE_CONNECTING) {
                targetCall.answer(VideoProfile.STATE_AUDIO_ONLY)
            }

            var waitedMs = 0
            while (targetCall.state != Call.STATE_ACTIVE && targetCall.state != Call.STATE_DISCONNECTED && waitedMs < 5000) {
                delay(100)
                waitedMs += 100
            }

            if (targetCall.state == Call.STATE_DISCONNECTED) {
                Log.w(TAG, "Call disconnected before reaching STATE_ACTIVE")
                handleCallDisconnected()
                return@launch
            }

            // Re-enforce silent screening state as soon as Telecom activates the call audio path
            enforceSilentScreeningAudioState()

            // Wait 500ms for VoLTE modem path setup
            delay(500)
            enforceSilentScreeningAudioState()
            executeVoicemailPipeline()
        }
    }

    /**
     * Forces Earpiece route, mutes STREAM_VOICE_CALL (silent in room), and mutes the phone's
     * physical microphone so the caller cannot hear the user during voicemail screening.
     */
    private fun enforceSilentScreeningAudioState() {
        try {
            setAudioRoute(CallAudioState.ROUTE_EARPIECE)
            setMuted(true)
            audioManager.isSpeakerphoneOn = false
            audioManager.isMicrophoneMute = true
            audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_MUTE, 0)
        } catch (t: Throwable) {
            Log.w(TAG, "Failed enforcing silent screening audio state: ${t.message}")
        }
    }

    private suspend fun executeVoicemailPipeline() {
        isVoicemailScreeningActive = true
        isGreetingFinished.set(false)
        chunksReceivedCount.set(0)

        TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.SCREENING_GREETING)
        updateInCallNotification()

        // Start transcriber session now so it is ready the instant the greeting finishes
        val number = TelecomCallManager.phoneNumber.value
        val name = TelecomCallManager.callerName.value
        val simName = TelecomCallManager.simDisplayName.value
        val simSlot = TelecomCallManager.simSlotIndex.value

        transcriber.startSession(
            callerName = name,
            phoneNumber = number,
            simDisplayName = simName,
            simSlotIndex = simSlot
        )

        val shizukuReady = withContext(Dispatchers.IO) {
            shizukuAudioController.updateShizukuStatus() && shizukuAudioController.ensureBound(2500L)
        }

        if (shizukuReady) {
            Log.i(TAG, "Shizuku VoicemailUserService is bound! Dispatching privileged uplink greeting + downlink capture.")
            try {
                val afd = GreetingGenerator.getGreetingAssetFileDescriptor(this)
                shizukuAudioController.startVoicemailSession(afd, audioChunkCallback)

                // Safety timeout in case binder callback is delayed
                greetingTimeoutJob?.cancel()
                greetingTimeoutJob = serviceScope.launch {
                    delay(8000L)
                    if (!isGreetingFinished.get() && isVoicemailScreeningActive) {
                        Log.w(TAG, "Privileged greeting timeout (8s) reached; ensuring SCREENING_LIVE_VOICEMAIL state.")
                        onGreetingPlaybackFinished()
                    }
                }
                return
            } catch (t: Throwable) {
                Log.e(TAG, "Failed starting Shizuku voicemail session, falling back", t)
            }
        }

        Log.w(TAG, "Shizuku unavailable or unbound! Using silent in-process telephony fallback.")
        playFallbackGreetingToTelephony()
    }

    /**
     * Fallback used ONLY when Shizuku is not running: routes greeting to TYPE_TELEPHONY / Earpiece
     * (never loudspeaker!) so it does not blast loudly in the room.
     */
    private fun playFallbackGreetingToTelephony() {
        try {
            enforceSilentScreeningAudioState()
            audioManager.setParameters("incall_music_enabled=true;incall_music_uplink=true;music_playback_during_call=true")

            val greetingFile = GreetingGenerator.getActiveGreetingFile(this)
            val attrs = AudioAttributes.Builder()
                .setUsage(AudioAttributes.USAGE_VOICE_COMMUNICATION)
                .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
                .build()

            val mp = MediaPlayer().apply {
                setAudioAttributes(attrs)
                setDataSource(greetingFile.absolutePath)
                val outputs = audioManager.getDevices(AudioManager.GET_DEVICES_OUTPUTS)
                val telephonyDevice = outputs.firstOrNull { it.type == AudioDeviceInfo.TYPE_TELEPHONY }
                    ?: outputs.firstOrNull { it.type == AudioDeviceInfo.TYPE_BUILTIN_EARPIECE }
                if (telephonyDevice != null) {
                    setPreferredDevice(telephonyDevice)
                }
                setVolume(1.0f, 1.0f)
                prepare()
            }
            inProcessGreetingPlayer = mp
            val dur = mp.duration

            mp.setOnCompletionListener {
                onGreetingPlaybackFinished()
                startInProcessDownlinkFallback()
            }
            mp.setOnErrorListener { _, _, _ ->
                onGreetingPlaybackFinished()
                startInProcessDownlinkFallback()
                true
            }

            val timeoutMs = if (dur > 0) (dur + 1000L) else 6500L
            greetingTimeoutJob?.cancel()
            greetingTimeoutJob = serviceScope.launch {
                delay(timeoutMs)
                if (!isGreetingFinished.get() && isVoicemailScreeningActive) {
                    onGreetingPlaybackFinished()
                    startInProcessDownlinkFallback()
                }
            }
            mp.start()
        } catch (t: Throwable) {
            Log.e(TAG, "Fallback greeting failed: ${t.message}", t)
            onGreetingPlaybackFinished()
            startInProcessDownlinkFallback()
        }
    }

    private fun onGreetingPlaybackFinished() {
        if (isGreetingFinished.getAndSet(true)) return
        greetingTimeoutJob?.cancel()
        greetingTimeoutJob = null
        stopInProcessGreetingFallback()

        if (!isVoicemailScreeningActive) return

        Log.i(TAG, "Greeting finished. Transitioning UI to SCREENING_LIVE_VOICEMAIL (mic muted, speaker off by default).")

        try {
            audioManager.setParameters("incall_music_enabled=false;incall_music_uplink=false")
            // Keep physical microphone strictly muted so caller cannot hear user!
            setMuted(true)
            audioManager.isMicrophoneMute = true

            // Respect user's "Listen Live" toggle (default is FALSE = completely silent in room)
            val listenLive = TelecomCallManager.isScreeningSpeakerOn.value
            setScreeningSpeakerEnabled(listenLive)
        } catch (t: Throwable) {
            Log.w(TAG, "Error configuring post-greeting audio state: ${t.message}")
        }

        TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.SCREENING_LIVE_VOICEMAIL)
        updateInCallNotification()
    }

    /**
     * Fallback downlink recorder used ONLY if Shizuku UID 2000 capture fails.
     * Attempts VOICE_DOWNLINK (3) and VOICE_CALL (4) with TYPE_TELEPHONY input while keeping the mic muted.
     */
    private fun startInProcessDownlinkFallback() {
        if (inProcessRecordJob != null || !isVoicemailScreeningActive) return
        inProcessRecordJob = serviceScope.launch(Dispatchers.IO) {
            val candidateSources = intArrayOf(
                3, // MediaRecorder.AudioSource.VOICE_DOWNLINK
                MediaRecorder.AudioSource.VOICE_CALL,
                MediaRecorder.AudioSource.VOICE_COMMUNICATION
            )
            val inputs = audioManager.getDevices(AudioManager.GET_DEVICES_INPUTS)
            val telephonyInput = inputs.firstOrNull { it.type == AudioDeviceInfo.TYPE_TELEPHONY }

            var record: AudioRecord? = null
            val minBuf = AudioRecord.getMinBufferSize(16000, AudioFormat.CHANNEL_IN_MONO, AudioFormat.ENCODING_PCM_16BIT)
            val bufferSize = max(minBuf, 3200 * 4)

            for (src in candidateSources) {
                try {
                    val candidate = AudioRecord(src, 16000, AudioFormat.CHANNEL_IN_MONO, AudioFormat.ENCODING_PCM_16BIT, bufferSize)
                    if (candidate.state == AudioRecord.STATE_INITIALIZED) {
                        if (telephonyInput != null) {
                            runCatching { candidate.setPreferredDevice(telephonyInput) }
                        }
                        candidate.startRecording()
                        if (candidate.recordingState == AudioRecord.RECORDSTATE_RECORDING) {
                            record = candidate
                            Log.i(TAG, "Fallback AudioRecord started with source=$src")
                            break
                        }
                    }
                    candidate.release()
                } catch (ignored: Throwable) {}
            }

            if (record == null) {
                Log.e(TAG, "Fallback AudioRecord could not open downlink source (Shizuku permission required).")
                return@launch
            }

            inProcessAudioRecord = record
            val buf = ShortArray(1600)
            try {
                while (isVoicemailScreeningActive) {
                    // Keep physical microphone muted so local voice is never recorded!
                    if (!audioManager.isMicrophoneMute) {
                        audioManager.isMicrophoneMute = true
                    }
                    val read = record.read(buf, 0, buf.size)
                    if (read > 0 && isGreetingFinished.get()) {
                        val copy = ShortArray(read)
                        for (i in 0 until read) {
                            copy[i] = (buf[i].toInt() * 2).coerceIn(-32768, 32767).toShort()
                        }
                        transcriber.submitPcmChunk(copy)
                    } else if (read < 0) {
                        break
                    }
                }
            } finally {
                runCatching {
                    record.stop()
                    record.release()
                }
                if (inProcessAudioRecord == record) inProcessAudioRecord = null
            }
        }
    }

    private fun stopInProcessGreetingFallback() {
        try {
            inProcessGreetingPlayer?.stop()
            inProcessGreetingPlayer?.release()
        } catch (ignored: Throwable) {}
        inProcessGreetingPlayer = null
    }

    private fun stopInProcessAudioRecord() {
        inProcessRecordJob?.cancel()
        inProcessRecordJob = null
        try {
            inProcessAudioRecord?.stop()
            inProcessAudioRecord?.release()
        } catch (ignored: Throwable) {}
        inProcessAudioRecord = null
    }

    /**
     * Handles user tapping "Pick Up" to take over the call and talk to the caller live.
     */
    fun takeOverVoicemailCall() {
        Log.i(TAG, "Taking over Live Voicemail session: stopping screening and unmuting user microphone.")
        isVoicemailScreeningActive = false
        stopInProcessGreetingFallback()
        stopInProcessAudioRecord()
        shizukuAudioController.stopVoicemailSession()

        // 1. Unmute hardware microphone & restore normal earpiece/call volume for live conversation
        setMuted(false)
        audioManager.isMicrophoneMute = false
        audioManager.isSpeakerphoneOn = false
        setAudioRoute(CallAudioState.ROUTE_EARPIECE)
        audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_UNMUTE, 0)
        val maxVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
        audioManager.setStreamVolume(AudioManager.STREAM_VOICE_CALL, (maxVol * 0.8f).toInt().coerceAtLeast(3), 0)

        // 2. Acquire proximity lock so ear proximity blanks display
        acquireProximityWakeLock()

        // 3. Update UI state to ActiveCall
        TelecomCallManager.setScreeningState(TelecomCallManager.CallScreeningState.ACTIVE_CALL)
        updateInCallNotification()

        // 4. Finalize voicemail record as picked up
        stopAndSaveVoicemailSession(isPickedUp = true)
    }

    /**
     * Toggles whether the user wants to hear the caller's voice out loud in the room while screening.
     * Microphone remains strictly muted either way.
     */
    fun setScreeningSpeakerEnabled(enabled: Boolean) {
        try {
            setMuted(true)
            audioManager.isMicrophoneMute = true
            shizukuAudioController.setScreeningSpeakerEnabled(enabled)

            if (enabled) {
                setAudioRoute(CallAudioState.ROUTE_SPEAKER)
                audioManager.isSpeakerphoneOn = true
                audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_UNMUTE, 0)
                val maxVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_VOICE_CALL)
                audioManager.setStreamVolume(AudioManager.STREAM_VOICE_CALL, (maxVol * 0.7f).toInt().coerceAtLeast(2), 0)
            } else {
                setAudioRoute(CallAudioState.ROUTE_EARPIECE)
                audioManager.isSpeakerphoneOn = false
                audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_MUTE, 0)
            }
            Log.i(TAG, "setScreeningSpeakerEnabled($enabled): micMuted=true, speaker=$enabled")
        } catch (t: Throwable) {
            Log.e(TAG, "Error toggling screening speaker: ${t.message}")
        }
    }

    fun stopAndSaveVoicemailSession(isPickedUp: Boolean = false, onComplete: (() -> Unit)? = null) {
        if (hasSavedCurrentSession.getAndSet(true)) {
            onComplete?.invoke()
            return
        }

        cancelAutoAnswerTimer()
        releaseProximityWakeLock()
        stopInProcessGreetingFallback()
        stopInProcessAudioRecord()
        shizukuAudioController.stopVoicemailSession()

        val wasScreening = isVoicemailScreeningActive || isScreeningSessionRunning.get()
        val wasGreetingPlaying = !isGreetingFinished.get()
        isVoicemailScreeningActive = false
        isScreeningSessionRunning.set(false)

        // Restore normal audio state when session ends
        setMuted(false)
        audioManager.isMicrophoneMute = false
        audioManager.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_UNMUTE, 0)

        TelecomCallManager.setScreeningState(
            if (isPickedUp) TelecomCallManager.CallScreeningState.ACTIVE_CALL
            else TelecomCallManager.CallScreeningState.DISCONNECTED
        )

        appScope.launch(NonCancellable) {
            try {
                if (wasGreetingPlaying && wasScreening && !isPickedUp) {
                    val number = TelecomCallManager.phoneNumber.value
                    val name = TelecomCallManager.callerName.value
                    val simName = TelecomCallManager.simDisplayName.value
                    val simSlot = TelecomCallManager.simSlotIndex.value

                    transcriber.stopSession(isPickedUp = false, saveToRepository = false)

                    val recordingsDir = java.io.File(filesDir, "recordings").apply { mkdirs() }
                    val cleanNumber = number.replace(Regex("[^0-9+]"), "").ifEmpty { "unknown" }
                    val timeStamp = java.text.SimpleDateFormat("yyyyMMdd_HHmmss", java.util.Locale.US).format(java.util.Date())
                    val finalWavFile = java.io.File(recordingsDir, "voicemail_${cleanNumber}_$timeStamp.wav")
                    try {
                        val silenceBytes = ByteArray(16000 * 2)
                        java.io.FileOutputStream(finalWavFile).use { fos ->
                            GreetingGenerator.writeWavHeader(
                                out = fos,
                                totalAudioLen = silenceBytes.size.toLong(),
                                sampleRate = 16000,
                                channels = 1,
                                bitsPerSample = 16
                            )
                            fos.write(silenceBytes)
                            fos.flush()
                        }
                    } catch (ignored: Throwable) {}

                    val entity = com.personal.voicemail.data.VoicemailEntity(
                        callerName = name,
                        phoneNumber = number,
                        simSlotIndex = simSlot,
                        simDisplayName = simName,
                        timestamp = System.currentTimeMillis(),
                        durationSeconds = 1,
                        transcript = "Caller hung up during greeting",
                        audioFilePath = finalWavFile.absolutePath,
                        isRead = false,
                        isPickedUp = false
                    )
                    VoicemailApplication.voicemailRepository.addVoicemail(entity)
                    postVoicemailCompletedNotification(name ?: number, "Caller hung up during greeting")

     * Phase A: Injects the greeting WAV into the cellular uplink TX while keeping the phone silent.
     * Strategy 1: AOSP Call Assistant / TYPE_TELEPHONY explicit routing (pure digital uplink injection).
     * Strategy 2: Qualcomm incall_music_uplink DSP mixer with earpiece routing so room hears nothing.
     */
    private fun executePhaseAGreetingInjection(greetingWavFd: AssetFileDescriptor?) {
        val am = audioManager ?: return
        Log.i(TAG, "Commencing Phase A: Silent Telephony Uplink Greeting Injection")

        val stagedTmpFile = File("/data/local/tmp/voicemail_greeting.wav")
        if (!stagedTmpFile.exists() || stagedTmpFile.length() < 32000) {
            greetingWavFd?.let { fd ->
                try {
                    val pfd = fd.parcelFileDescriptor
                    val inStream = if (pfd != null) ParcelFileDescriptor.AutoCloseInputStream(pfd.dup()) else fd.createInputStream()
                    FileOutputStream(stagedTmpFile).use { fos ->
                        inStream.copyTo(fos)
                        fos.flush()
                    }
                    stagedTmpFile.setReadable(true, false)
                } catch (t: Throwable) {
                    Log.w(TAG, "Could not cache greeting to /data/local/tmp: ${t.message}")
                }
            }
        }

            readWavFileSampleRate(stagedTmpFile)
        } else {
            readWavSampleRate(greetingWavFd)
        }

        // Keep hardware mic muted so room noise is never mixed into greeting
        runCatching {
            am.isSpeakerphoneOn = false
            am.isMicrophoneMute = true
            am.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_MUTE, 0)
            am.setParameters(QUALCOMM_PARAM_INCALL_MUSIC_ON)

        val telephonyOutputDevice = outputs.firstOrNull { it.type == AudioDeviceInfo.TYPE_TELEPHONY }
        val earpieceDevice = outputs.firstOrNull { it.type == AudioDeviceInfo.TYPE_BUILTIN_EARPIECE }

            .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
            .setSampleRate(targetSampleRate)
            .setChannelMask(AudioFormat.CHANNEL_OUT_MONO)

            targetSampleRate,
            AudioFormat.CHANNEL_OUT_MONO,
            AudioFormat.ENCODING_PCM_16BIT

        fun createAudioTrack(attrs: AudioAttributes, preferredDevice: AudioDeviceInfo?): AudioTrack? {

                    .setAudioAttributes(attrs)
                    .setAudioFormat(audioFormat)
                    .setBufferSizeInBytes(trackBufferSize)
                    .setTransferMode(AudioTrack.MODE_STREAM)

                shellContext?.let { ctx ->
                    runCatching {
                        val m = AudioTrack.Builder::class.java.getDeclaredMethod("setContext", Context::class.java)
                        m.isAccessible = true
                        m.invoke(builder, ctx)
                    }
                }
                val candidate = builder.build()
                if (candidate.state == AudioTrack.STATE_INITIALIZED) {
                    if (preferredDevice != null) {
                        runCatching { candidate.setPreferredDevice(preferredDevice) }
                    }
                    candidate
                } else {
                    candidate.release()
                    null
                }
            } catch (t: Throwable) {
                Log.w(TAG, "Failed creating AudioTrack: ${t.message}")
                null
            }
        }

        val activeTracks = mutableListOf<AudioTrack>()

        // Track 1: USAGE_CALL_ASSISTANT (17) routed directly to TYPE_TELEPHONY

            t1.setVolume(1.0f)
            t1.play()
            activeTracks.add(t1)
            Log.i(TAG, "Phase A Track 1 (CALL_ASSISTANT) playing on routedDevice=${t1.routedDevice?.type}")

            .setUsage(AudioAttributes.USAGE_MEDIA)
            .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
            .build()

    private fun readWavFileSampleRate(file: File): Int {
        var sampleRate = DEFAULT_SAMPLE_RATE_HZ
        try {
            FileInputStream(file).use { fis ->
                val header = ByteArray(44)
                val read = fis.read(header)
                if (read >= 28) {
                    val rate = ByteBuffer.wrap(header, 24, 4).order(ByteOrder.LITTLE_ENDIAN).int
                    if (rate in 8000..48000) {
                        sampleRate = rate
                    }
                }
            }
        } catch (t: Throwable) {
            sampleRate = DEFAULT_SAMPLE_RATE_HZ
        }
        return sampleRate
    }

    /**
     * Phase B: Privileged Downlink Capture.
     * Prioritizes VOICE_DOWNLINK (3) and TYPE_TELEPHONY input so ONLY the remote caller's voice
     * is captured and transcribed, never the local user's microphone.
     */
    private fun executePhaseBDownlinkCapture(callback: IAudioChunkCallback) {
        val am = audioManager ?: run {
            Log.e(TAG, "AudioManager not available during Phase B downlink capture")
            return
        }

        Log.i(TAG, "Commencing Phase B: Privileged Telephony Downlink Capture")

        val inputs = am.getDevices(AudioManager.GET_DEVICES_INPUTS)
        val telephonyInputDevice = inputs.firstOrNull { it.type == AudioDeviceInfo.TYPE_TELEPHONY }
        if (telephonyInputDevice != null) {
            Log.i(TAG, "Telephony input device located: ID=${telephonyInputDevice.id}")
        }

        val minBufSize = AudioRecord.getMinBufferSize(
            DEFAULT_SAMPLE_RATE_HZ,
            AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT
        )
        val recordBufferSize = max(minBufSize, CHUNK_SIZE_SAMPLES * 2 * 4)

        fun createAndStartRecord(source: Int, routeToTelephonyDevice: Boolean): AudioRecord? {
            return try {
                val attrBuilder = AudioAttributes.Builder()
                val presetApplied = runCatching {
                    val m = AudioAttributes.Builder::class.java.getDeclaredMethod(
                        "setInternalCapturePreset",
                        Int::class.javaPrimitiveType
                    )
                    m.isAccessible = true
                    m.invoke(attrBuilder, source)
                    true
                }.getOrDefault(false)

                val recBuilder = AudioRecord.Builder()
                shellContext?.let { ctx ->
                    runCatching {
                        val mCtx = AudioRecord.Builder::class.java.getDeclaredMethod(
                            "setContext",
                            Context::class.java
                        )
                        mCtx.isAccessible = true
                        mCtx.invoke(recBuilder, ctx)
                    }
                }
                if (presetApplied) {
                    runCatching {
                        val mAttr = AudioRecord.Builder::class.java.getDeclaredMethod(
                            "setAudioAttributes",
                            AudioAttributes::class.java
                        )
                        mAttr.isAccessible = true
                        mAttr.invoke(recBuilder, attrBuilder.build())
                    }.onFailure {
                        recBuilder.setAudioSource(source)
                    }
                } else {
                    recBuilder.setAudioSource(source)
                }

                recBuilder.setAudioFormat(
                    AudioFormat.Builder()
                        .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
                        .setSampleRate(DEFAULT_SAMPLE_RATE_HZ)
                        .setChannelMask(AudioFormat.CHANNEL_IN_MONO)
                        .build()
                ).setBufferSizeInBytes(recordBufferSize)

                val rec = recBuilder.build()
                if (rec.state != AudioRecord.STATE_INITIALIZED) {
                    rec.release()
                    return null
                }
                if (routeToTelephonyDevice && telephonyInputDevice != null) {
                    runCatching { rec.setPreferredDevice(telephonyInputDevice) }
                }
                rec.startRecording()
                if (rec.recordingState != AudioRecord.RECORDSTATE_RECORDING) {
                    rec.release()
                    return null
                }
                Log.i(TAG, "Phase B AudioRecord started: source=$source, presetApplied=$presetApplied, routeTelephony=$routeToTelephonyDevice, routedDevice=${rec.routedDevice?.type}")
                rec
            } catch (t: Throwable) {
                Log.w(TAG, "Failed creating Phase B AudioRecord(source=$source, route=$routeToTelephonyDevice): ${t.message}")
                null
            }
        }

        // Strictly prioritize telephony downlink sources (VOICE_DOWNLINK=3, VOICE_CALL=4, or explicit TYPE_TELEPHONY device)
        // Because physical mic is muted (isMicrophoneMute = true), VOICE_CALL only contains the downlink leg!
        val candidates = mutableListOf(
            Pair(AUDIO_SOURCE_VOICE_DOWNLINK, true),
            Pair(AUDIO_SOURCE_VOICE_DOWNLINK, false),
            Pair(AUDIO_SOURCE_VOICE_CALL, true),
            Pair(AUDIO_SOURCE_VOICE_CALL, false)
        )
        if (telephonyInputDevice != null) {
            candidates.add(Pair(MediaRecorder.AudioSource.VOICE_RECOGNITION, true))
            candidates.add(Pair(MediaRecorder.AudioSource.VOICE_COMMUNICATION, true))
        }

        var candidateIndex = 0
        var activeRecord: AudioRecord? = null

        while (candidateIndex < candidates.size) {
            val cand = candidates[candidateIndex]
            val rec = createAndStartRecord(cand.first, cand.second)
            if (rec != null) {
                activeRecord = rec
                break
            }
            candidateIndex++
        }

        if (activeRecord == null) {
            val errorMsg = "Unable to initialize privileged downlink AudioRecord in UID 2000."
            Log.e(TAG, errorMsg)
            callback.onError(errorMsg)
            return
        }

        currentAudioRecord = activeRecord
        isRecordingActive.set(true)

        val recordingThread = Thread({
            Process.setThreadPriority(Process.THREAD_PRIORITY_URGENT_AUDIO)

            val pcmByteBuffer = ByteArray(CHUNK_SIZE_SAMPLES * 2)
            var chunkIndex = 0
            var consecutiveZeroChunks = 0

            Log.i(TAG, "Phase B downlink capture loop active (candidate=${candidates[candidateIndex]})")

            while (isSessionActive.get() && isRecordingActive.get()) {
                val record = activeRecord ?: break
                val currentCandidate = candidates[candidateIndex]
                val bytesRead = record.read(pcmByteBuffer, 0, pcmByteBuffer.size)

                if (bytesRead > 0) {
                    var peakAmp = 0
                    var i = 0
                    while (i < bytesRead - 1) {
                        val sample = (pcmByteBuffer[i].toInt() and 0xFF or (pcmByteBuffer[i + 1].toInt() shl 8)).toShort()
                        val absVal = kotlin.math.abs(sample.toInt())
                        if (absVal > peakAmp) peakAmp = absVal
                        i += 2
                    }

                    chunkIndex++
                    if (chunkIndex <= 20 || chunkIndex % 15 == 0) {
                        Log.d(TAG, "Downlink RX chunk #$chunkIndex source=${currentCandidate.first} route=${currentCandidate.second} peak=$peakAmp")
                    }

                    if (peakAmp == 0) {
                        consecutiveZeroChunks++
                    } else {
                        consecutiveZeroChunks = 0
                    }

                    // Only switch candidate if we get 25 consecutive exact-zero chunks (2.5 seconds of dead hardware stream)
                    if (consecutiveZeroChunks >= 25 && candidateIndex < candidates.lastIndex) {
                        Log.w(TAG, "Downlink source $currentCandidate returned 25 zero chunks; switching to next downlink candidate...")
                        runCatching {
                            record.stop()
                            record.release()
                        }
                        var switched = false
                        while (candidateIndex < candidates.lastIndex) {
                            candidateIndex++
                            val nextCand = candidates[candidateIndex]
                            val newRecord = createAndStartRecord(nextCand.first, nextCand.second)
                            if (newRecord != null) {
                                activeRecord = newRecord
                                currentAudioRecord = newRecord
                                consecutiveZeroChunks = 0
                                switched = true
                                break
                            }
                        }
                        if (!switched) break
                        continue
                    }

                    val chunk = if (bytesRead == pcmByteBuffer.size) {
                        pcmByteBuffer.clone()
                    } else {
                        pcmByteBuffer.copyOf(bytesRead)
                    }

                    try {
                        callback.onPcmChunkReceived(chunk)
                    } catch (re: RemoteException) {
                        Log.e(TAG, "RemoteException in onPcmChunkReceived; stopping capture", re)
                        break
                    }
                } else if (bytesRead < 0) {
                    Log.w(TAG, "AudioRecord read error $bytesRead on $currentCandidate")
                    break
                }
            }

            Log.i(TAG, "Phase B downlink capture loop terminated. Total chunks: $chunkIndex")
        }, "LiveVoicemail-DownlinkLoop")

        activeRecordingThread = recordingThread
        recordingThread.start()
    }

    @Synchronized
    override fun stopVoicemailSession() {
        Log.i(TAG, "stopVoicemailSession requested.")
        isSessionActive.set(false)
        isRecordingActive.set(false)

        try {
            activeRecordingThread?.interrupt()
            activeRecordingThread = null
        } catch (ignored: Throwable) {}

        try {
            currentAudioRecord?.let {
                if (it.recordingState == AudioRecord.RECORDSTATE_RECORDING) {
                    it.stop()
                }
                it.release()
            }
        } catch (t: Throwable) {
            Log.w(TAG, "Error stopping AudioRecord", t)
        } finally {
            currentAudioRecord = null
        }

        try {
            currentAudioTrack?.let {
                if (it.playState == AudioTrack.PLAYSTATE_PLAYING) {
                    it.stop()
                }
                it.release()
            }
        } catch (t: Throwable) {
            Log.w(TAG, "Error stopping AudioTrack", t)
        } finally {
            currentAudioTrack = null
        }

        audioManager?.let { am ->
            runCatching {
                am.setParameters(QUALCOMM_PARAM_INCALL_MUSIC_OFF)
                am.isMicrophoneMute = false
                am.isSpeakerphoneOn = false
                am.adjustStreamVolume(AudioManager.STREAM_VOICE_CALL, AudioManager.ADJUST_UNMUTE, 0)
                if (savedVoiceCallVolume > 0) {
                    am.setStreamVolume(AudioManager.STREAM_VOICE_CALL, savedVoiceCallVolume, 0)
                }
                if (savedMusicVolume >= 0) {
                    am.setStreamVolume(AudioManager.STREAM_MUSIC, savedMusicVolume, 0)
                }
            }
        }

        try {
            activeWorkerExecutor?.shutdownNow()
            activeWorkerExecutor = null
        } catch (ignored: Throwable) {}

        Log.i(TAG, "stopVoicemailSession completed cleanly.")
    }

    override fun destroy() {
        Log.i(TAG, "destroy() invoked on Shizuku UserService.")
        stopVoicemailSession()
        Process.killProcess(Process.myPid())
    }

private fun querySimAccounts(context: Context): List<SimAccountOption> {
    val results = ArrayList<SimAccountOption>()
    val telecomManager = context.getSystemService(TelecomManager::class.java)
    val subscriptionManager = context.getSystemService(SubscriptionManager::class.java)

    if (ContextCompat.checkSelfPermission(context, android.Manifest.permission.READ_PHONE_STATE) != PackageManager.PERMISSION_GRANTED) {
        return listOf(SimAccountOption(0, "SIM 1", null))
    }

    try {
        val activeSubs = subscriptionManager?.activeSubscriptionInfoList ?: emptyList<SubscriptionInfo>()
        val callHandles = telecomManager?.callCapablePhoneAccounts ?: emptyList<PhoneAccountHandle>()

        if (activeSubs.isNotEmpty()) {
            for (sub in activeSubs) {
                val matchedHandle = callHandles.firstOrNull { it.id == sub.subscriptionId.toString() || it.id == sub.iccId }
                val name = sub.displayName?.toString()?.ifBlank { sub.carrierName?.toString() } ?: "SIM ${sub.simSlotIndex + 1}"
                results.add(
                    SimAccountOption(
                        slotIndex = sub.simSlotIndex,
                        displayName = name,
                        phoneAccountHandle = matchedHandle
                    )
                )
            }
        } else {
            results.add(SimAccountOption(0, "SIM 1", callHandles.firstOrNull()))
        }
    } catch (e: Exception) {
        results.add(SimAccountOption(0, "SIM 1", null))
    }

    return results.sortedBy { it.slotIndex }
}

private fun placeOutgoingCall(
    context: Context,
    number: String,
    accountHandle: PhoneAccountHandle?
) {
    val telecomManager = context.getSystemService(TelecomManager::class.java) ?: return
    val uri = Uri.fromParts("tel", number, null)
    val extras = Bundle().apply {
        if (accountHandle != null) {
            putParcelable(TelecomManager.EXTRA_PHONE_ACCOUNT_HANDLE, accountHandle)
        }
    }

    try {
        if (ContextCompat.checkSelfPermission(context, android.Manifest.permission.CALL_PHONE) == PackageManager.PERMISSION_GRANTED) {
            telecomManager.placeCall(uri, extras)
        }
    } catch (e: Exception) {
        android.util.Log.e("DialpadTab", "Failed to place call to $number", e)
    }
}

private suspend fun fetchCallLogs(context: Context): List<CallLogEntry> = withContext(Dispatchers.IO) {
    if (ContextCompat.checkSelfPermission(context, android.Manifest.permission.READ_CALL_LOG) != PackageManager.PERMISSION_GRANTED) {
        return@withContext emptyList()
    }

    val result = ArrayList<CallLogEntry>()
    val projection = arrayOf(
        CallLog.Calls._ID,
        CallLog.Calls.NUMBER,
        CallLog.Calls.CACHED_NAME,
        CallLog.Calls.DATE,
        CallLog.Calls.DURATION,
        CallLog.Calls.TYPE,
        CallLog.Calls.PHONE_ACCOUNT_ID
    )

    try {
        context.contentResolver.query(
            CallLog.Calls.CONTENT_URI,
            projection,
            null,
            null,
            "${CallLog.Calls.DATE} DESC LIMIT 100"
        )?.use { cursor ->
            val colId = cursor.getColumnIndex(CallLog.Calls._ID)
            val colNumber = cursor.getColumnIndex(CallLog.Calls.NUMBER)
            val colName = cursor.getColumnIndex(CallLog.Calls.CACHED_NAME)
            val colDate = cursor.getColumnIndex(CallLog.Calls.DATE)
            val colDuration = cursor.getColumnIndex(CallLog.Calls.DURATION)
            val colType = cursor.getColumnIndex(CallLog.Calls.TYPE)
            val colSimId = cursor.getColumnIndex(CallLog.Calls.PHONE_ACCOUNT_ID)

            while (cursor.moveToNext()) {
                val id = cursor.getLong(colId)
                val number = cursor.getString(colNumber) ?: "Unknown"
                val name = cursor.getString(colName)
                val date = cursor.getLong(colDate)
                val duration = cursor.getLong(colDuration)
                val type = cursor.getInt(colType)
                val simId = if (colSimId != -1) cursor.getString(colSimId) else null

                result.add(
                    CallLogEntry(
                        id = id,
                        number = number,
                        name = name,
                        date = date,
                        durationSec = duration,
                        type = type,
                        simAccountId = simId
                    )
                )
            }
        }
    } catch (e: Exception) {
        android.util.Log.e("RecentsTab", "Error reading call logs", e)
    }

    result
}

private fun formatLogDate(dateMs: Long): String {
    val date = Date(dateMs)
    val now = Date()
    val cal1 = java.util.Calendar.getInstance().apply { time = date }
    val cal2 = java.util.Calendar.getInstance().apply { time = now }

    return if (cal1.get(java.util.Calendar.YEAR) == cal2.get(java.util.Calendar.YEAR) &&
        cal1.get(java.util.Calendar.DAY_OF_YEAR) == cal2.get(java.util.Calendar.DAY_OF_YEAR)
    ) {
        SimpleDateFormat("h:mm a", Locale.getDefault()).format(date)
    } else {
        SimpleDateFormat("MMM d", Locale.getDefault()).format(date)
    }
}

private fun checkDefaultDialerRole(context: Context): Boolean {
    val roleManager = context.getSystemService(RoleManager::class.java) ?: return false
    return roleManager.isRoleHeld(RoleManager.ROLE_DIALER)
}

private fun requestDefaultDialerRole(
    context: Context,
    launcher: androidx.activity.result.ActivityResultLauncher<Intent>
) {
    val roleManager = context.getSystemService(RoleManager::class.java) ?: return
    if (roleManager.isRoleAvailable(RoleManager.ROLE_DIALER)) {
        val intent = roleManager.createRequestRoleIntent(RoleManager.ROLE_DIALER)
        launcher.launch(intent)
    }
}

private fun checkBatteryOptimizationExempt(context: Context): Boolean {
    val powerManager = context.getSystemService(PowerManager::class.java) ?: return false
    return powerManager.isIgnoringBatteryOptimizations(context.packageName)
}

private fun requestBatteryExemption(context: Context) {
    try {
        val intent = Intent(Settings.ACTION_REQUEST_IGNORE_BATTERY_OPTIMIZATIONS).apply {
            data = Uri.parse("package:${context.packageName}")
        }
        context.startActivity(intent)
    } catch (e: Exception) {
        val intent = Intent(Settings.ACTION_IGNORE_BATTERY_OPTIMIZATION_SETTINGS)
        context.startActivity(intent)
    }
}

private fun checkRuntimePermissions(context: Context): Boolean {
    val permissions = listOf(
        android.Manifest.permission.CALL_PHONE,
        android.Manifest.permission.READ_PHONE_STATE,
        android.Manifest.permission.READ_CALL_LOG,
        android.Manifest.permission.READ_CONTACTS,
        android.Manifest.permission.RECORD_AUDIO
    )
    return permissions.all {
        ContextCompat.checkSelfPermission(context, it) == android.content.pm.PackageManager.PERMISSION_GRANTED
    }
}

private fun checkAccessibilityServiceEnabled(context: Context): Boolean {
    val expected = "${context.packageName}/${com.personal.voicemail.service.VoicemailAccessibilityService::class.java.canonicalName}"
    val expectedShort = "${context.packageName}/.service.VoicemailAccessibilityService"
    val enabledServices = Settings.Secure.getString(
        context.contentResolver,
        Settings.Secure.ENABLED_ACCESSIBILITY_SERVICES
    ) ?: return false
    return enabledServices.contains(expected) || enabledServices.contains(expectedShort)
}

@Composable
private fun VoicemailAudioPlayerRow(
    audioPath: String,
    durationSeconds: Int,
    isPlaying: Boolean,
    onPlayToggle: () -> Unit,
    onDelete: () -> Unit
) {
    val context = LocalContext.current
    var mediaPlayer by remember { mutableStateOf<MediaPlayer?>(null) }
    var currentPositionMs by remember { mutableIntStateOf(0) }
    var totalDurationMs by remember { mutableIntStateOf(durationSeconds * 1000) }
    var isDraggingSlider by remember { mutableStateOf(false) }

    val mainHandler = remember { Handler(Looper.getMainLooper()) }

    DisposableEffect(audioPath, isPlaying) {
        if (isPlaying) {
            try {
                val mp = MediaPlayer().apply {
                    setDataSource(audioPath)
                    prepare()
                    totalDurationMs = duration
                    start()
                    setOnCompletionListener {
                        onPlayToggle()
                    }
                }
                mediaPlayer = mp

                val progressRunnable = object : Runnable {
                    override fun run() {
                        val active = mediaPlayer ?: return
                        val playing = runCatching { active.isPlaying }.getOrDefault(false)
                        if (playing && !isDraggingSlider) {
                            runCatching { currentPositionMs = active.currentPosition }
                            mainHandler.postDelayed(this, 100)
                        }
                    }
                }
                mainHandler.post(progressRunnable)
            } catch (e: Exception) {
                Log.e("VoicemailPlayer", "Error initializing MediaPlayer for $audioPath", e)
            }
        } else {
            mainHandler.removeCallbacksAndMessages(null)
            mediaPlayer?.let {
                runCatching {
                    if (it.isPlaying) it.stop()
                    it.release()
                }
            }
            mediaPlayer = null
            currentPositionMs = 0
        }

        onDispose {
            mainHandler.removeCallbacksAndMessages(null)
            mediaPlayer?.let {
                runCatching {
                    if (it.isPlaying) it.stop()
                    it.release()
                }
            }
            mediaPlayer = null
        }
    }

    Row(
        modifier = Modifier.fillMaxWidth(),
        verticalAlignment = Alignment.CenterVertically
    ) {
        // Play/Pause circular button
        Box(
            modifier = Modifier
                .size(40.dp)
                .clip(CircleShape)
                .background(if (isPlaying) AccentBlue else GlassSurface)
                .border(1.dp, GlassBorder, CircleShape)
                .clickable { onPlayToggle() },
            contentAlignment = Alignment.Center
        ) {

private fun formatDate(epochMs: Long): String {
    val date = Date(epochMs)
    val now = Date()
    val cal1 = java.util.Calendar.getInstance().apply { time = date }
    val cal2 = java.util.Calendar.getInstance().apply { time = now }

    return if (cal1.get(java.util.Calendar.YEAR) == cal2.get(java.util.Calendar.YEAR) &&
        cal1.get(java.util.Calendar.DAY_OF_YEAR) == cal2.get(java.util.Calendar.DAY_OF_YEAR)
    ) {
        SimpleDateFormat("h:mm a", Locale.getDefault()).format(date)
    } else {
        SimpleDateFormat("MMM d, h:mm a", Locale.getDefault()).format(date)
    }
}

private fun formatDuration(seconds: Int): String {
    val m = seconds / 60
    val s = seconds % 60
    return String.format("%02d:%02d", m, s)
}

enum class MainTab(val title: String, val icon: ImageVector) {
    VOICEMAIL("Voicemail", Icons.Default.Voicemail),
    RECENTS("Recents", Icons.Default.History),
    DIALPAD("Keypad", Icons.Default.Dialpad)

class MainActivity : ComponentActivity() {

    private var dialNumberState = mutableStateOf("")
    private var selectedTabState = mutableIntStateOf(0)

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        handleDialIntent(intent)

        setContent {
            PersonalVoicemailTheme {
                MainAppScreen(
                    selectedTabIndex = selectedTabState.intValue,
                    dialNumber = dialNumberState.value,
                    onTabSelected = { index -> selectedTabState.intValue = index },
                    onCallFromRecents = { number ->
                        dialNumberState.value = number
                        selectedTabState.intValue = MainTab.DIALPAD.ordinal
                    }
                )
            }
        }
    }

    override fun onResume() {
        super.onResume()
        com.personal.voicemail.VoicemailApplication.voicemailRepository.reloadFromDisk()
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        setIntent(intent)
        handleDialIntent(intent)
    }

    private fun handleDialIntent(intent: Intent?) {
        val data: Uri? = intent?.data
        if (data != null && data.scheme == "tel") {
            val number = data.schemeSpecificPart ?: ""
            if (number.isNotBlank()) {
                dialNumberState.value = number
                selectedTabState.intValue = MainTab.DIALPAD.ordinal
            }
        }
    }
}

@Composable
fun MainAppScreen(
    selectedTabIndex: Int,
    dialNumber: String,
    onTabSelected: (Int) -> Unit,
    onCallFromRecents: (String) -> Unit
) {
    val tabs = MainTab.values()

    Scaffold(
        modifier = Modifier
            .fillMaxSize()
            .background(DarkBackground)
            .statusBarsPadding(),
        bottomBar = {
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .navigationBarsPadding()
                    .padding(horizontal = 16.dp, vertical = 8.dp)
                    .clip(RoundedCornerShape(24.dp))
                    .background(Color(0xCC1A1A22))
                    .border(1.dp, GlassBorder, RoundedCornerShape(24.dp))
            ) {
                NavigationBar(
                    containerColor = Color.Transparent,
                    tonalElevation = 0.dp,
                    modifier = Modifier.fillMaxWidth()
                ) {
                    tabs.forEachIndexed { index, tab ->
                        val isSelected = selectedTabIndex == index
                        NavigationBarItem(
                            selected = isSelected,
                            onClick = { onTabSelected(index) },
                            icon = {
                                Icon(
                                    imageVector = tab.icon,
                                    contentDescription = tab.title,
                                    modifier = Modifier.size(24.dp)
                                )
                            },
                            label = {
                                Text(
                                    text = tab.title,
                                    fontSize = 11.sp,
                                    color = if (isSelected) AccentBlue else TextSecondary
                                )
                            },
                            colors = NavigationBarItemDefaults.colors(
                                selectedIconColor = AccentBlue,
                                selectedTextColor = AccentBlue,
                                unselectedIconColor = TextSecondary,
                                unselectedTextColor = TextSecondary,
                                indicatorColor = Color(0x33007AFF)
                            )
                        )
                    }
                }
            }
        },
        containerColor = DarkBackground
    ) { innerPadding ->
        Column(
            modifier = Modifier
                .fillMaxSize()
                .padding(innerPadding)
        ) {
            // System Readiness & Permissions Banner (Top of screen)
            SystemReadinessBanner(
                modifier = Modifier.padding(horizontal = 16.dp, vertical = 8.dp)
            )

            // Active Tab Content
            Box(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
            ) {
                when (selectedTabIndex) {
                    MainTab.VOICEMAIL.ordinal -> {
                        VoicemailInboxTab()
                    }
                    MainTab.RECENTS.ordinal -> {
                        RecentsTab(onCallNumber = onCallFromRecents)
                    }
                    MainTab.DIALPAD.ordinal -> {
                        DialpadTab(initialNumber = dialNumber)
                    }
                }
            }
        }
    }

import android.app.Application

class VoicemailApplication : Application() {

    companion object {
        const val TAG = "VoicemailApp"
        const val CHANNEL_INCALL_ID = "channel_live_voicemail_incall"
        const val CHANNEL_VOICEMAIL_NOTIFICATIONS_ID = "channel_voicemail_recordings"

        private lateinit var instance: VoicemailApplication

        fun getAppContext(): Context = instance.applicationContext

        lateinit var shizukuController: com.personal.voicemail.service.ShizukuAudioController
            private set
        lateinit var voicemailRepository: com.personal.voicemail.data.VoicemailRepository
            private set
        lateinit var transcriber: com.personal.voicemail.recognition.LiveVoicemailTranscriber
            private set
    }

    override fun onCreate() {
        super.onCreate()
        instance = this

        // 1. Bypass Hidden API Restrictions for Android 12 (API 31) through Android 15 (API 35)
        exemptHiddenApis()

        // 2. Initialize Process-Wide Shared Singletons
        shizukuController = com.personal.voicemail.service.ShizukuAudioController.getInstance(this)
        voicemailRepository = com.personal.voicemail.data.VoicemailRepository.getInstance(this)
        transcriber = com.personal.voicemail.recognition.LiveVoicemailTranscriber.getInstance(this)

        shizukuController.requestShizukuPermission()
        shizukuController.bindUserService()

        // 3. Initialize High-Priority Notification Channels for InCallService and Voicemail Alerts
        createNotificationChannels()

        // 4. Guarantee default greeting WAV file is synthesized on disk
        try {
            val greeting = com.personal.voicemail.audio.GreetingGenerator.getActiveGreetingFile(this)
            Log.i(TAG, "Greeting WAV initialized at: ${greeting.absolutePath} (${greeting.length()} bytes)")
        } catch (t: Throwable) {
            Log.e(TAG, "Failed to initialize greeting WAV", t)
        }

        // 5. Trigger spoken voice greeting TTS synthesis in background
        com.personal.voicemail.audio.GreetingGenerator.generateVoiceGreetingTts(this)

        Log.i(TAG, "VoicemailApplication initialized. PID: ${android.os.Process.myPid()}, UID: ${android.os.Process.myUid()}")
    }

    private fun exemptHiddenApis() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) {
            try {
                val success = HiddenApiBypass.addHiddenApiExemptions("L")
                Log.i(TAG, "HiddenApiBypass applied successfully in Application process: $success")
            } catch (t: Throwable) {
                Log.e(TAG, "Failed to apply HiddenApiBypass in Application process", t)
            }
        }
    }

    private fun createNotificationChannels() {
        val notificationManager = getSystemService(NotificationManager::class.java) ?: return

        // Channel for Incoming Call Full-Screen Intent and Foreground InCallService
        val inCallChannel = NotificationChannel(
            CHANNEL_INCALL_ID,
            getString(R.string.notification_channel_incall),
            NotificationManager.IMPORTANCE_HIGH
        ).apply {
            description = "Maintains the active call and live screening status bar notifications"
            lockscreenVisibility = NotificationManager.IMPORTANCE_HIGH
            setSound(null, null) // InCallService handles telephony audio routing directly
            enableVibration(false)
        }

        // Channel for New Voicemail Transcript Notifications
        val voicemailChannel = NotificationChannel(
            CHANNEL_VOICEMAIL_NOTIFICATIONS_ID,
            getString(R.string.notification_channel_voicemail),
            NotificationManager.IMPORTANCE_DEFAULT
        ).apply {
            description = "Alerts the user when a caller completes a live voicemail recording"
            enableVibration(true)
        }

        notificationManager.createNotificationChannel(inCallChannel)
        notificationManager.createNotificationChannel(voicemailChannel)
    }
}
